            let ticking = false;
            
            function updateImages() {
                const scrollY = window.scrollY;
                const windowHeight = window.innerHeight;
                const maxScroll = windowHeight * 0.8;
                
                const progress = Math.min(scrollY / maxScroll, 1);
                const images = document.querySelectorAll('.image-container');
                const scales = [4, 6, 5, 7, 5, 8, 4, 6];
                
                images.forEach((image, index) => {
                    const scale = 1 + (scales[index] * progress);
                    const opacity = 1 - (progress * 1.2);
                    
                    image.style.transform = `scale(${scale})`;
                    image.style.opacity = Math.max(0, opacity);
                });
                
                ticking = false;
            }
            
            function requestTick() {
                if (!ticking) {
                    requestAnimationFrame(updateImages);
                    ticking = true;
                }
            }
            
            window.addEventListener('scroll', requestTick);
            updateImages();

            // Tab functionality
            function switchToTab(index, shouldScroll = true) {
                tabBtns.forEach(btn => btn.classList.remove('active'));
                tabContents.forEach(content => {
                    content.classList.remove('active');
                    content.style.display = 'none';
                });
                
                const targetBtn = tabBtns[index];
                const targetTab = targetBtn.getAttribute('data-tab');
                targetBtn.classList.add('active');
                
                let targetContentId;
                if (targetTab === 'numbers') {
                    targetContentId = 'numbers-section';
                } else if (targetTab === 'blog') {
                    targetContentId = 'stories-section';
                } else if (targetTab === 'cv') {
                    targetContentId = 'resume-section';
                }
                
                const targetContent = document.getElementById(targetContentId);
                if (targetContent) {
                    targetContent.style.display = 'block';
                    targetContent.classList.add('active');
                    
                    if (shouldScroll) {
                        setTimeout(() => {
                            const offsetTop = targetContent.offsetTop - 100;
                            window.scrollTo({
                                top: offsetTop,
                                behavior: 'smooth'
                            });
                        }, 200);
                    }
                }
            }
            
            if (tabBtns.length > 0 && tabContents.length > 0) {
                switchToTab(0, false);
            }
            
            tabBtns.forEach((btn, index) => {
                btn.addEventListener('click', () => {
                    switchToTab(index, true);
                });
            });
        });
