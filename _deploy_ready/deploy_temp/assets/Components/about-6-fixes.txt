3. Update WebGL Particle Code (in the JavaScript section)
Find the particle creation code and replace with:
javascript// Create constellation particles (80 particles, subtle)
const particlesGeometry = new THREE.BufferGeometry();
const particlesCount = window.innerWidth <= 767 ? 40 : 80;
const posArray = new Float32Array(particlesCount * 3);

for(let i = 0; i < particlesCount * 3; i++) {
    posArray[i] = (Math.random() - 0.5) * 15;
}

particlesGeometry.setAttribute('position', new THREE.BufferAttribute(posArray, 3));

// Mix of Coral and Cloud Dancer colors
const colors = new Float32Array(particlesCount * 3);
for(let i = 0; i < particlesCount; i++) {
    const isCoralParticle = Math.random() > 0.5;
    if(isCoralParticle) {
        // Coral: #F96F6E
        colors[i * 3] = 0.976;
        colors[i * 3 + 1] = 0.435;
        colors[i * 3 + 2] = 0.431;
    } else {
        // Cloud Dancer: #F3EFE0
        colors[i * 3] = 0.953;
        colors[i * 3 + 1] = 0.937;
        colors[i * 3 + 2] = 0.878;
    }
}
particlesGeometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

const particlesMaterial = new THREE.PointsMaterial({
    size: 0.012,
    vertexColors: true,
    transparent: true,
    opacity: 0.08,
    blending: THREE.AdditiveBlending
});

const particlesMesh = new THREE.Points(particlesGeometry, particlesMaterial);
scene.add(particlesMesh);
