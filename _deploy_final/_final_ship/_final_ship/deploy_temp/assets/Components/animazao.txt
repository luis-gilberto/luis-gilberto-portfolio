          Object.assign(learnMoreCTA.style, {
            position: 'fixed',
            left: '50%',

            gap: '0.5rem',

            backdropFilter: 'blur(10px)',
            WebkitBackdropFilter: 'blur(10px)',
            border: '1px solid rgba(46, 211, 198, 0.2)',
            borderRadius: '8px',
            color: 'rgba(46, 211, 198, 0.7)',
            textDecoration: 'none',
            fontFamily: 'Inter, sans-serif',
            fontWeight: '500',
            fontSize: '0.875rem',
            letterSpacing: '0.02em',
            opacity: '0',
            transform: 'translateX(-50%) translateY(10px)',
            transition: 'all 0.4s cubic-bezier(0.16, 1, 0.3, 1)',
            pointerEvents: 'none',
            boxShadow: '0 0 0 rgba(46, 211, 198, 0)',
            textShadow: '0 0 0 rgba(46, 211, 198, 0)'
          });
          
          // Mobile responsive
          if (window.innerWidth <= 768) {
            learnMoreCTA.style.left = '50%';
            learnMoreCTA.style.bottom = '6rem';
            learnMoreCTA.style.transform = 'translateX(-50%) translateY(10px)';
            learnMoreCTA.style.padding = '0.625rem 1rem';
            learnMoreCTA.style.fontSize = '0.8125rem';
          }
          
          // Hover effects
          learnMoreCTA.addEventListener('mouseenter', function() {
            this.style.background = 'rgba(46, 211, 198, 0.25)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.6)';
            this.style.color = 'rgba(46, 211, 198, 1)';
            this.style.transform = 'translateX(-50%) translateY(0) scale(1.02)';
            this.style.boxShadow = '0 0 30px rgba(46, 211, 198, 0.3), 0 0 60px rgba(46, 211, 198, 0.1)';
            this.style.textShadow = '0 0 20px rgba(46, 211, 198, 0.5)';
            
            const svg = this.querySelector('.deeper-arrow');
            if (svg) svg.style.transform = 'translateX(4px)';
          });
          
          learnMoreCTA.addEventListener('mouseleave', function() {
            this.style.background = 'rgba(46, 211, 198, 0.08)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.2)';
            this.style.color = 'rgba(46, 211, 198, 0.7)';
            this.style.transform = 'translateX(-50%) translateY(0) scale(1)';
            this.style.boxShadow = '0 0 20px rgba(46, 211, 198, 0.15)';
            this.style.textShadow = '0 0 10px rgba(46, 211, 198, 0.3)';
            
            const svg = this.querySelector('.deeper-arrow');
            if (svg) svg.style.transform = 'translateX(0)';
          });

          // ... rest of your code (flash effects, etc.) ...

          // Reveal (make sure to use centered transform)

            learnMoreCTA.style.boxShadow = '0 0 20px rgba(46, 211, 198, 0.15)';
            learnMoreCTA.style.textShadow = '0 0 10px rgba(46, 211, 198, 0.3)';

          // Hide (make sure to use centered transform)
          setTimeout(() => {
            learnMoreCTA.style.opacity = '0';

              if (learnMoreCTA.parentNode) {
                learnMoreCTA.remove();
              }
            }, 400);
