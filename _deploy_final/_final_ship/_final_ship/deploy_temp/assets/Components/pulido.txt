## 4) ADD REVEAL SCRIPT (ONCE PER PAGE)
Target: before </body>
If an element with id="lg-reveal-script" does NOT exist, append the following:

<script id="lg-reveal-script">
  (() => {
    const els = document.querySelectorAll('.reveal');
    if (!('IntersectionObserver' in window)) {
      els.forEach(el => el.classList.add('in'));
      // ensure underline appears without IO
      document.querySelectorAll('.tagline').forEach(t => t.classList.add('in'));
      return;
    }
    const io = new IntersectionObserver(entries => {
      entries.forEach(e => {
        if (e.isIntersecting){
          e.target.classList.add('in');
          if (e.target.classList.contains('tagline')) {
            // ensure underline draw
            e.target.classList.add('in');
          }
          io.unobserve(e.target);
        }
      });
    }, { rootMargin: '0px 0px -10% 0px', threshold: 0.1 });
    els.forEach(el => io.observe(el));
  })();
</script>
