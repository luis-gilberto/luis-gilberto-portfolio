      // ========================================================================
      // ADD "LEARN MORE ABOUT THE HUB" CTA TO ANIMATION

      setTimeout(() => {
        // Find the hub loading container
        const hubContainer = document.querySelector('.hub-loading-container');
        
        if (hubContainer) {
          // Create the "Learn More" CTA
          const learnMoreCTA = document.createElement('a');
          learnMoreCTA.href = 'https://luis-gilberto.com/TheHub/system/';
          learnMoreCTA.className = 'hub-animation-learn-more';
          learnMoreCTA.innerHTML = `
            <span>Learn more about The Hub</span>
            <svg viewBox="0 0 24 24" fill="none" width="18" height="18">
              <path d="M5 12h14M12 5l7 7-7 7" stroke="currentColor" stroke-width="2" stroke-linecap="round"/>
            </svg>
          `;
          
          // Add inline styles to the CTA
          Object.assign(learnMoreCTA.style, {
            position: 'fixed',
            left: '2rem',
            bottom: '2rem',
            zIndex: '10001',
            display: 'inline-flex',
            alignItems: 'center',
            gap: '0.75rem',
            padding: '1rem 1.5rem',
            background: 'rgba(46, 211, 198, 0.15)',
            backdropFilter: 'blur(20px)',
            WebkitBackdropFilter: 'blur(20px)',
            border: '1px solid rgba(46, 211, 198, 0.4)',
            borderRadius: '12px',
            color: '#2ED3C6',
            textDecoration: 'none',
            fontFamily: 'Inter, sans-serif',
            fontWeight: '600',
            fontSize: '0.95rem',
            opacity: '0',
            transform: 'translateX(-20px)',
            transition: 'all 0.4s cubic-bezier(0.16, 1, 0.3, 1)',
            pointerEvents: 'none',
            boxShadow: '0 8px 32px rgba(46, 211, 198, 0.15)'
          });
          
          // Mobile responsive styles
          if (window.innerWidth <= 768) {
            learnMoreCTA.style.left = '1rem';
            learnMoreCTA.style.bottom = '1rem';
            learnMoreCTA.style.padding = '0.75rem 1rem';
            learnMoreCTA.style.fontSize = '0.875rem';
          }
          
          // Hover effects
          learnMoreCTA.addEventListener('mouseenter', function() {
            this.style.background = 'rgba(46, 211, 198, 0.25)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.6)';
            this.style.transform = 'translateX(0) scale(1.02)';
            this.style.boxShadow = '0 12px 40px rgba(46, 211, 198, 0.25)';
            
            // Animate arrow
            const svg = this.querySelector('svg');
            if (svg) svg.style.transform = 'translateX(4px)';
          });
          
          learnMoreCTA.addEventListener('mouseleave', function() {
            this.style.background = 'rgba(46, 211, 198, 0.15)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.4)';
            this.style.transform = 'translateX(0) scale(1)';
            this.style.boxShadow = '0 8px 32px rgba(46, 211, 198, 0.15)';
            
            // Reset arrow
            const svg = this.querySelector('svg');
            if (svg) svg.style.transform = 'translateX(0)';
          });
          
          // Add transition to SVG
          const svg = learnMoreCTA.querySelector('svg');
          if (svg) {
            svg.style.transition = 'transform 0.3s ease';
          }
          
          // Track navigation (prevent animation replay when returning)
          learnMoreCTA.addEventListener('click', function(e) {
            sessionStorage.setItem('hubNavigated', 'true');
          });
          
          // Add CTA to the hub container
          hubContainer.appendChild(learnMoreCTA);
          
          // Show CTA after 9 seconds (3 seconds before animation ends)
          setTimeout(() => {
            learnMoreCTA.style.opacity = '1';
            learnMoreCTA.style.transform = 'translateX(0)';
            learnMoreCTA.style.pointerEvents = 'auto';
          }, 9000);
          
          // Hide CTA when animation completes (at 12 seconds)
          setTimeout(() => {
            learnMoreCTA.style.opacity = '0';
            learnMoreCTA.style.transform = 'translateX(-20px)';
            setTimeout(() => {
              if (learnMoreCTA.parentNode) {
                learnMoreCTA.remove();
              }
            }, 400);
          }, 12000);
        }
      }, 200);
