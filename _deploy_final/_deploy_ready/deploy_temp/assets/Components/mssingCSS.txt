## ALSO ADD JAVASCRIPT
Add this JavaScript right before your closing `</body>` tag (after your existing script):

```html
<script>
// Microsoft Logo Rotator
class CompactLogoRotator {
    constructor() {
        this.currentIndex = 0;
        this.logos = [
            { name: 'Microsoft Copilot', file: '/assets/microsoft-copilot.png' },
            { name: 'Microsoft Edge', file: '/assets/microsoft-edge.png' },
            { name: 'Microsoft 365', file: '/assets/microsoft-365.png' },
            { name: 'Microsoft Teams', file: '/assets/microsoft-teams.png' },
            { name: 'Office 365', file: '/assets/office-365.png' },
            { name: 'Surface Pro', file: '/assets/surface.png' },
            { name: 'Microsoft Band', file: '/assets/microsoft-band.png' }
        ];
        
        this.init();
    }

    init() {
        this.createLogoElements();
        this.showLogo(0);
        this.startRotation();
    }

    createLogoElements() {
        const display = document.getElementById('logoDisplay');
        if (!display) return;
        
        this.logos.forEach((logo, index) => {
            const logoItem = document.createElement('div');
            logoItem.className = 'logo-item';
            logoItem.innerHTML = `<img src="${logo.file}" alt="${logo.name}" loading="lazy">`;
            display.appendChild(logoItem);
        });
    }

    showLogo(index) {
        const logoItems = document.querySelectorAll('.logo-item');
        logoItems.forEach(item => item.classList.remove('active'));
        if (logoItems[index]) logoItems[index].classList.add('active');

        const counter = document.getElementById('logoCounter');
        if (counter) counter.textContent = `${index + 1} / ${this.logos.length}`;
        this.currentIndex = index;
    }

    nextLogo() {
        const nextIndex = (this.currentIndex + 1) % this.logos.length;
        this.showLogo(nextIndex);
    }

    startRotation() {
        setInterval(() => this.nextLogo(), 3000);
    }
}

// Initialize when page loads
document.addEventListener('DOMContentLoaded', function() {
    setTimeout(() => {
        new CompactLogoRotator();
    }, 500);
});
</script>
```

This adds all the missing CSS that styles your Microsoft Legacy section and footer. The HTML structure you have is correct - it just needed the styling rules to make it look proper.
