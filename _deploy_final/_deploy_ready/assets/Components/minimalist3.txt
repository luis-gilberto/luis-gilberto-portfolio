    <script>
        /* =============================
           🧠 Cursor follower logic
           ============================= */
        const cursor = document.getElementById('cursor');
        let rafId;
        let targetX = 0, targetY = 0;
        let x = 0, y = 0;
        const lerp = (a,b,t)=>a+(b-a)*t;

        const activate = () => cursor.classList.add('cursor--active');
        const deactivate = () => cursor.classList.remove('cursor--active');

        window.addEventListener('pointermove', (e)=>{
            targetX = e.clientX - 120; // center the 240px circle
            targetY = e.clientY - 120;
            activate();
            if(!rafId){
                const loop = () =>{
                    x = lerp(x, targetX, 0.18);
                    y = lerp(y, targetY, 0.18);
                    cursor.style.transform = `translate3d(${x}px, ${y}px, 0)`;
                    rafId = requestAnimationFrame(loop);
                };
                rafId = requestAnimationFrame(loop);
            }
        });
        window.addEventListener('pointerleave', deactivate);

        /* =============================
           🌫️ Parallax orbs on scroll
           ============================= */
        const orbs = document.querySelectorAll('.orb');
        window.addEventListener('scroll', () => {
            const scrollY = window.scrollY || window.pageYOffset;
            orbs.forEach(orb => {
                const speed = parseFloat(orb.dataset.speed || 0.5);
                orb.style.transform = `translate3d(0, ${-scrollY * speed * 0.08}px, 0)`;
            });
        });

        /* =============================
           🔗 Enhanced scroll behavior
           ============================= */
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const target = document.querySelector(this.getAttribute('href'));
                if (target) {
                    target.scrollIntoView({
                        behavior: 'smooth',
                        block: 'start'
                    });
                }
            });
        });

        /* =============================
           ⌨️ Keyboard navigation for sections
           ============================= */
        document.addEventListener('keydown', function(e) {
            if (e.key === 'ArrowDown' || e.key === 'PageDown') {
                e.preventDefault();
                scrollToNextSection();
            } else if (e.key === 'ArrowUp' || e.key === 'PageUp') {
                e.preventDefault();
                scrollToPrevSection();
            }
        });

        function scrollToNextSection() {
            const sections = document.querySelectorAll('.section');
            const currentScroll = window.scrollY;
            
            for (let i = 0; i < sections.length; i++) {
                const sectionTop = sections[i].offsetTop;
                if (sectionTop > currentScroll + 50) {
                    sections[i].scrollIntoView({ behavior: 'smooth' });
                    break;
                }
            }
        }

        function scrollToPrevSection() {
            const sections = document.querySelectorAll('.section');
            const currentScroll = window.scrollY;
            
            for (let i = sections.length - 1; i >= 0; i--) {
                const sectionTop = sections[i].offsetTop;
                if (sectionTop < currentScroll - 50) {
                    sections[i].scrollIntoView({ behavior: 'smooth' });
                    break;
                }
            }
        }
    </script>
