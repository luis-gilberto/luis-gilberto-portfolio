PROBLEM 3: Timer counting too fast (microseconds instead of seconds)

Find the timer update function in script.js and replace it with this corrected version:

```javascript
// FIX: Timer Update Function
let timerStartTime = null;
let timerInterval = null;

function startAssessmentTimer() {
  timerStartTime = Date.now();
  
  // Clear any existing interval
  if (timerInterval) {
    clearInterval(timerInterval);
  }
  
  // Update every second (1000ms)
  timerInterval = setInterval(updateTimer, 1000);
}

function updateTimer() {
  if (!timerStartTime) return;
  
  const elapsed = Math.floor((Date.now() - timerStartTime) / 1000); // Convert to seconds
  const minutes = Math.floor(elapsed / 60);
  const seconds = elapsed % 60;
  
  const timerElement = document.getElementById('assessment-timer');
  if (timerElement) {
    timerElement.textContent = 
      `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
  }
}

function stopAssessmentTimer() {
  if (timerInterval) {
    clearInterval(timerInterval);
    timerInterval = null;
  }
}

// Make sure to call startAssessmentTimer() when assessment begins
// and stopAssessmentTimer() when it ends or exits
```

Then find where the assessment modal opens (likely in an event listener) and add:

```javascript
// When assessment starts, add this:

// When assessment exits, add this:
stopAssessmentTimer();
```
