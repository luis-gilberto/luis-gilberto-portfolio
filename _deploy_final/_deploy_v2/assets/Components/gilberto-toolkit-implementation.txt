    class TOCManager {
        constructor() {
            this.tocToggle = document.getElementById('toc-toggle');
            this.tocList = document.getElementById('toc-list');
            this.tocLinks = document.querySelectorAll('.toc-link');
            this.sections = ['hero', 'services', 'investment', 'testimonials', 'contact'];
            this.isCollapsed = false;
            
            this.init();
        }
        
        init() {
            this.bindEvents();
            this.updateActiveSection();
            this.handleResponsiveBehavior();
        }
        
        bindEvents() {
            // Toggle TOC visibility
            if (this.tocToggle) {
                this.tocToggle.addEventListener('click', () => {
                    this.toggleTOC();
                });
            }
            
            // Smooth scrolling for TOC links
            this.tocLinks.forEach(link => {
                link.addEventListener('click', (e) => {
                    e.preventDefault();
                    const targetId = link.getAttribute('href').substring(1);
                    this.scrollToSection(targetId);
                    this.setActiveLink(link);
                });
            });
            
            // Update active section on scroll with throttling
            let scrollTimeout;
            window.addEventListener('scroll', () => {
                if (scrollTimeout) {
                    clearTimeout(scrollTimeout);
                }
                scrollTimeout = setTimeout(() => {
                    this.updateActiveSection();
                }, 50);
            });
            
            // Handle window resize
            window.addEventListener('resize', () => {
                this.handleResponsiveBehavior();
            });
        }
        
        toggleTOC() {
            this.isCollapsed = !this.isCollapsed;
            this.tocToggle.setAttribute('aria-expanded', !this.isCollapsed);
            
            if (this.isCollapsed) {
                this.tocList.style.display = 'none';
            } else {
                this.tocList.style.display = 'block';
            }
        }
        
        scrollToSection(sectionId) {
            const section = document.getElementById(sectionId);
            if (section) {
                const yOffset = -20; // Offset for better positioning
                const y = section.getBoundingClientRect().top + window.pageYOffset + yOffset;
                
                window.scrollTo({
                    top: y,
                    behavior: 'smooth'
                });
            }
        }
        
        setActiveLink(activeLink) {
            this.tocLinks.forEach(link => link.classList.remove('active'));
            activeLink.classList.add('active');
        }
        
        updateActiveSection() {
            const scrollPosition = window.scrollY + 100;
            let currentSection = this.sections[0]; // Default to first section
            
            this.sections.forEach(sectionId => {
                const section = document.getElementById(sectionId);
                if (section) {
                    const sectionTop = section.offsetTop;
                    const sectionHeight = section.offsetHeight;
                    
                    if (scrollPosition >= sectionTop && scrollPosition < sectionTop + sectionHeight) {
                        currentSection = sectionId;
                    }
                }
            });
            
            const correspondingLink = document.querySelector(`[href="#${currentSection}"]`);
            if (correspondingLink && !correspondingLink.classList.contains('active')) {
                this.setActiveLink(correspondingLink);
            }
        }
        
        handleResponsiveBehavior() {
            const isMobile = window.innerWidth <= 768;
            const tocNav = document.getElementById('toc-nav');
            
            if (tocNav) {
