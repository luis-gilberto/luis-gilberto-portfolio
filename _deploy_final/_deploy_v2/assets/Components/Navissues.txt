## STEP 3: JAVASCRIPT IMPLEMENTATION

### A) Create imc-navigation.js (conflict-safe)

```javascript
/* ===========================================
   IMC NAVIGATION CONTROLLER
   Conflict-safe implementation for ToC pages
   =========================================== */

class IMCNavigation {
  constructor() {
    this.mobileMenuBtn = document.getElementById('imcMobileMenuBtn');
    this.mobileOverlay = document.getElementById('imcMobileOverlay');
    this.mobileClose = document.getElementById('imcMobileClose');
    this.crossNavItems = document.querySelectorAll('.imc-cross-nav-item');
    
    this.init();
  }

  init() {
    this.bindEvents();
    this.updateActiveStates();
    this.handleToCCompatibility();
  }

  bindEvents() {
    // Mobile menu events - using unique event namespace
    if (this.mobileMenuBtn) {
      this.mobileMenuBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        this.openMobileMenu();
      });
    }

    if (this.mobileClose) {
      this.mobileClose.addEventListener('click', (e) => {
        e.stopPropagation();
        this.closeMobileMenu();
      });
    }

    if (this.mobileOverlay) {
      this.mobileOverlay.addEventListener('click', (e) => {
        if (e.target === this.mobileOverlay) {
          this.closeMobileMenu();
        }
      });
    }

    // Cross-nav interactive effects
    this.crossNavItems.forEach(item => {
      item.addEventListener('mouseenter', this.handleHoverIn.bind(this));
      item.addEventListener('mouseleave', this.handleHoverOut.bind(this));
    });

    // Escape key to close mobile menu
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        this.closeMobileMenu();
      }
    });

    // Prevent conflicts with ToC scroll events
    this.preventToCConflicts();
  }

  openMobileMenu() {
    this.mobileOverlay?.classList.add('active');
    document.body.style.overflow = 'hidden';
  }

  closeMobileMenu() {
    this.mobileOverlay?.classList.remove('active');
    document.body.style.overflow = '';
  }

  handleHoverIn(e) {
    if (!e.currentTarget.classList.contains('active')) {
      e.currentTarget.style.transform = 'translateY(-3px)';
    }
  }

  handleHoverOut(e) {
    if (!e.currentTarget.classList.contains('active')) {
      e.currentTarget.style.transform = 'translateY(0)';
    } else {
      e.currentTarget.style.transform = 'translateY(-1px)';
    }
  }

  updateActiveStates() {
    const currentPath = window.location.pathname;
    
    // Clear all active states
    this.crossNavItems.forEach(item => item.classList.remove('active'));
    document.querySelectorAll('.imc-mobile-nav a').forEach(item => item.classList.remove('active'));

    // Set active state based on current page
    let activeFound = false;
    
    if (currentPath.includes('/scopeiq-wizard')) {
      this.setActive('[data-page="scopeiq"]', 'ScopeIQ Wizard');
      activeFound = true;
    } else if (currentPath.includes('/engagement-toolkit')) {
      this.setActive('[data-page="toolkit"]', 'Engagement Toolkit');
      activeFound = true;
    } else if (currentPath.includes('/IMCServices')) {
      this.setActive('[data-page="services"]', 'IMC Services');
      activeFound = true;
    }

    // Fallback for services page
    if (!activeFound) {
      this.setActive('[data-page="services"]', 'IMC Services');
    }
  }

  setActive(selector, mobileLabel) {
    // Cross navigation
    const crossNavItem = document.querySelector(selector);
    if (crossNavItem) {
      crossNavItem.classList.add('active');
      crossNavItem.style.transform = 'translateY(-1px)';
    }

    // Mobile navigation
    const mobileNavItem = document.querySelector(`.imc-mobile-nav a[href*="${mobileLabel.toLowerCase().replace(' ', '-')}"]`);
    if (mobileNavItem) {
      mobileNavItem.classList.add('active');
    }
  }

  handleToCCompatibility() {
    // Prevent ToC events from interfering with navigation
    const tocElements = document.querySelectorAll('.toc-sidebar, .toc-overlay, .table-of-contents');
    
    tocElements.forEach(element => {
      // Ensure ToC events don't bubble up to navigation
      element.addEventListener('click', (e) => {
        e.stopPropagation();
      });
      
      // Lower z-index if needed
      const computedStyle = window.getComputedStyle(element);
      const zIndex = parseInt(computedStyle.zIndex);
      if (zIndex >= 1000) {
        element.style.zIndex = '899';
      }
    });
  }

  preventToCConflicts() {
    // Create a namespace for IMC navigation events
    const imcEventNamespace = 'imc-nav';
    
    // Override any existing conflicting event listeners
    document.addEventListener('scroll', (e) => {
      // Allow ToC scroll handling but prevent conflicts
      if (e.target.closest('.imc-cross-nav, .imc-top-nav')) {
        e.preventDefault();
      }
    }, { passive: false, capture: true });
  }

  // Public method to manually update navigation (useful for SPA-like behavior)
  refresh() {
    this.updateActiveStates();
  }
}

// Initialize when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
  window.imcNavigation = new IMCNavigation();
});

// Export for potential module usage
if (typeof module !== 'undefined' && module.exports) {
  module.exports = IMCNavigation;
}
```

## STEP 5: TESTING & VALIDATION

### A) Test Checklist for each page:

1. **Navigation State Management**
   - [ ] Correct active state highlighting
   - [ ] Proper breadcrumb display
   - [ ] Cross-navigation works between pages

2. **Mobile Responsiveness**
   - [ ] Mobile menu opens/closes properly
   - [ ] Bottom navigation adapts to mobile layout
   - [ ] No horizontal scrolling issues

3. **ToC Compatibility (Services & Toolkit pages)**
   - [ ] ToC still functions normally
   - [ ] No event conflicts between ToC and navigation
   - [ ] Proper z-index stacking
   - [ ] Smooth scrolling works for both systems

4. **Performance**
   - [ ] No console errors
   - [ ] Smooth animations
   - [ ] Fast page transitions

### B) Common Issues & Solutions:

**Issue: ToC and navigation conflict**
Solution: Use event namespacing and stopPropagation() in the JavaScript

**Issue: Mobile menu doesn't close**
Solution: Check that all event listeners are properly bound with unique IDs

**Issue: Active states not updating**
Solution: Verify the updateActiveStates() method is called on page load

**Issue: Z-index conflicts**
Solution: Use CSS custom properties for consistent z-index values

## FINAL NOTES FOR TRAE:

- All class names are prefixed with "imc-" to prevent conflicts
- Event handlers use stopPropagation() to prevent ToC interference  
- Z-index values are controlled via CSS custom properties
- Mobile menu uses unique IDs to prevent conflicts
- JavaScript is wrapped in a class for namespace protection
- Active state management is automatic but can be manually refreshed

The ScopeIQ Wizard page already has this navigation implemented, so focus on the Services Overview and Engagement Toolkit pages.
