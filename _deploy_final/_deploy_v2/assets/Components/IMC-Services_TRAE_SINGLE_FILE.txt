Add to scripts/imc-services.js (end of file is fine):
// Optional: pricing toggle safety (no-op if control hidden)
(function(){
  var toggle = document.querySelector('[data-action="toggle-pricing"]');
  if (!toggle) return;
  var basePrices = document.querySelectorAll('.base-price');
  var detailed = document.querySelectorAll('.detailed-price');
  toggle.addEventListener('click', function(){
    var detailedVisible = detailed[0] && getComputedStyle(detailed[0]).display !== 'none';
    detailed.forEach(function(el){ el.style.display = detailedVisible ? 'none' : 'inline'; });
    basePrices.forEach(function(el){ el.style.display = detailedVisible ? 'inline' : 'none'; });
  });
})();
