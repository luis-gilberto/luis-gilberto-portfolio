// Save or update assessment session
async function saveAssessmentSession() {
    try {
        // Calculate current intelligence score
        const currentScore = calculateCurrentScore();
        
        const sessionData = {
            session_name: `Assessment - ${new Date().toLocaleDateString()} ${new Date().toLocaleTimeString()}`,
            assessment_type: selectedService || 'comprehensive',
            assessment_area: selectedService || 'comprehensive', 
            current_question: currentQuestionIndex + 1,
            total_questions: currentQuestions?.length || 0,
            responses: assessmentData,
            intelligence_score: currentScore,
            service_recommendation: getServiceRecommendation(currentScore),
            confidence_percentage: calculateConfidence(currentScore),
            status: (currentQuestionIndex >= (currentQuestions?.length || 0) - 1) ? 'completed' : 'in-progress',
            consultant_notes: document.getElementById('consultantNotes')?.value || '',
            completion_time: Math.round((Date.now() - assessmentStartTime) / 60000) // minutes
        };

        // Add client_id if we have one selected
        if (currentClientId) {
            sessionData.client_id = currentClientId;
        }

        if (currentSessionId) {
            // Update existing session
            const { data, error } = await supabase
                .from('assessment_sessions')
                .update(sessionData)
                .eq('id', currentSessionId)
                .select();
            
            if (error) throw error;
            console.log('Assessment session updated:', currentSessionId);
            return data[0];
        } else {
            // Create new session
            const { data, error } = await supabase
                .from('assessment_sessions')
                .insert([sessionData])
                .select();
            
            if (error) throw error;
            
            if (data && data[0]) {
                currentSessionId = data[0].id;
                console.log('New assessment session created:', currentSessionId);
                return data[0];
            }
        }
    } catch (error) {
        console.error('Error saving assessment session:', error);
        
        // Fallback to localStorage
        const fallbackData = {
            sessionId: currentSessionId || 'local_' + Date.now(),
            responses: assessmentData,
            timestamp: new Date().toISOString(),
            score: calculateCurrentScore()
        };
        localStorage.setItem('strategyiq_backup', JSON.stringify(fallbackData));
        console.log('Assessment saved to localStorage as backup');
        
        return null;
    }
}

// Load assessment session
async function loadAssessmentSession(sessionId) {

// Calculate current assessment score
function calculateCurrentScore() {
    if (!assessmentData || Object.keys(assessmentData).length === 0) {
        return 0;
    }
    
    // Use existing calculateIntelligenceScore function if it exists
    if (typeof calculateIntelligenceScore === 'function') {
        return calculateIntelligenceScore();
    }
    
    // Fallback calculation
    let totalScore = 0;
    let questionCount = 0;
    
    Object.values(assessmentData).forEach(response => {
        if (typeof response === 'number') {
            totalScore += response;
            questionCount++;
        }
    });
    
    return questionCount > 0 ? (totalScore / questionCount) * 5 : 0; // Scale to 0-25
}

// Get service recommendation based on score
function getServiceRecommendation(score) {
    if (score >= 20) return 'Strategic Campaigns';
    if (score >= 15) return 'Brand Intelligence';
    if (score >= 12) return 'Growth Partnership';
    if (score >= 8) return 'Quick-Start Sprint';
    if (score >= 5) return 'Creative Strategy';
    return 'Founder\'s Ally';
}

// Calculate recommendation confidence
function calculateConfidence(score) {
    // Higher scores = higher confidence, with some randomness for realism
    const baseConfidence = Math.min(95, Math.max(65, (score * 3.5) + 15));
    const randomVariation = (Math.random() - 0.5) * 10; // ±5% variation
    return Math.round(baseConfidence + randomVariation);
}

                    await selectClient(e.target.value);

            // Display client context information

// Select and load client data
async function selectClient(clientId) {
    try {
        const { data, error } = await supabase
            .from('clients')
            .select('*')
            .eq('id', clientId)
            .single();
        
        if (error) throw error;
        
        currentClientId = clientId;

// Update client display in UI
function updateClientDisplay(client) {
    // Update any client info displays
    const clientNameDisplay = document.getElementById('selectedClientName');
    const clientCompanyDisplay = document.getElementById('selectedClientCompany');
    const clientProjectDisplay = document.getElementById('selectedClientProject');

    if (contextPanel) {

                .map(point => `<li>${point}</li>`)

        const budgetDisplay = document.getElementById('clientBudget');

        const timelineDisplay = document.getElementById('clientTimeline');
        if (timelineDisplay) timelineDisplay.textContent = client.timeline || 'Not specified';

}
