6.1 ADD MOBILE-SPECIFIC INTERACTIONS
INSERT this JavaScript before the closing </body> tag:

```javascript
<script id="mobile-enhancements">
// Mobile-specific enhancements
document.addEventListener('DOMContentLoaded', function() {
    // Prevent iOS zoom on input focus
    const viewportMeta = document.querySelector('meta[name="viewport"]');
    if (viewportMeta) {
        viewportMeta.setAttribute('content', 'width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no');
    }
    
    // Improve touch scrolling on iOS
    document.addEventListener('touchstart', function() {}, {passive: true});
    
    // Close mobile menu when clicking outside
    document.addEventListener('click', function(e) {
        const headerNav = document.getElementById('headerNav');
        const mobileToggle = document.getElementById('mobileMenuToggle');
        
        if (headerNav && headerNav.classList.contains('mobile-active')) {
            if (!headerNav.contains(e.target) && !mobileToggle.contains(e.target)) {
                headerNav.classList.remove('mobile-active');
                const spans = mobileToggle.querySelectorAll('span');
                spans[0].style.transform = 'none';
                spans[1].style.opacity = '1';
                spans[2].style.transform = 'none';
            }
        }
    });
    
    // Add haptic feedback for iOS
    if ('vibrate' in navigator) {
        document.querySelectorAll('.btn, .nav-item, .service-card').forEach(el => {
            el.addEventListener('touchstart', () => {
                navigator.vibrate(10);
            });
        });
    }
});
</script>
```
