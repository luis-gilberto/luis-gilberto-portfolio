            // Calculate visible cards and total cards
            const getVisibleCards = () => {
                const trackWidth = document.querySelector('.fs-track').offsetWidth;
                const cardWidth = 280 + 16; // card width + gap
                return Math.floor(trackWidth / cardWidth);
            };
            
            const getTotalCards = () => {
                return document.querySelectorAll('.fs-cards-container .fs-card').length;
            };

            function updateFilmstrip() {
                const visibleCards = getVisibleCards();
                const totalCards = getTotalCards();
                const maxIndex = Math.max(0, totalCards - visibleCards);
                
                // Clamp currentIndex within bounds
                currentIndex = Math.max(0, Math.min(currentIndex, maxIndex));
                
                const cardWidth = 280 + 16; // card width + gap
                const offset = -currentIndex * cardWidth;
                cardsContainer.style.transform = `translateX(${offset}px)`;
                
                // Update progress bar

                // Update button states
                prevBtn.disabled = currentIndex <= 0;
                nextBtn.disabled = currentIndex >= maxIndex;
                
                // Update button opacity
                prevBtn.style.opacity = currentIndex <= 0 ? '0.3' : '1';
                nextBtn.style.opacity = currentIndex >= maxIndex ? '0.3' : '1';

                if (currentIndex > 0) {

                const totalCards = getTotalCards();

            updateFilmstrip();
            
            // Update on window resize
            window.addEventListener('resize', updateFilmstrip);

            // Premium video management - ensure seamless autoplay and looping
            function initializePremiumVideos() {
                const videos = document.querySelectorAll('video');
                
                videos.forEach(video => {
                    // Prevent any user interaction
                    video.addEventListener('click', function(e) {
                        e.preventDefault();
                        e.stopPropagation();
                    });
                    
                    video.addEventListener('dblclick', function(e) {
                        e.preventDefault();
                        e.stopPropagation();
                    });
                    
                    // Disable context menu
                    video.addEventListener('contextmenu', function(e) {
                        e.preventDefault();
                        return false;
                    });
                    
                    // Ensure continuous loop without interruption
                    video.addEventListener('ended', function() {
                        this.currentTime = 0;
                        this.play().catch(() => {});
                    });
                    
                    // Prevent seeking
                    video.addEventListener('seeking', function() {
                        this.currentTime = 0;
                    });
                    
                    // Force play if paused (except in modal)
                    video.addEventListener('pause', function() {
                        if (!this.closest('.modal-content')) {
                            setTimeout(() => {
                                this.play().catch(() => {});
                            }, 100);
                        }
                    });
                    
                    // Start playing immediately
                    video.play().catch(() => {});
                });
            }
            
            // Initialize premium videos
            initializePremiumVideos();

            // Modal functionality
            const modal = document.getElementById('timeline-modal');
            
            function openTimelineModal(era) {
                if (!modal) return;
                
                const data = timelineData[era] || {};
                
                // Populate modal content
                document.getElementById('modal-title').textContent = data.title || '';
                document.getElementById('modal-quote').textContent = data.quote ? `"${data.quote}"` : '';
                document.getElementById('modal-description').innerHTML = `<p>${data.description || ''}</p>`;
                
                // Populate achievements
                const achievementsList = document.querySelector('#modal-achievements ul');
                achievementsList.innerHTML = '';
                (data.achievements || []).forEach(achievement => {
                    const li = document.createElement('li');
                    li.innerHTML = achievement;
                    achievementsList.appendChild(li);
                });
                
                // Populate brands supported
                const brandsSupportedContainer = document.querySelector('#modal-brands-supported .brands-supported-image');
                const brandsSupportedSection = document.querySelector('#modal-brands-supported');
                if (data.brandsSupported && data.brandsSupported.image) {
                    brandsSupportedContainer.innerHTML = '';
                    const img = document.createElement('img');
                    img.src = data.brandsSupported.image.src;
                    img.alt = data.brandsSupported.image.alt;
                    img.onerror = function() { brandsSupportedSection.style.display = 'none'; };
                    brandsSupportedContainer.appendChild(img);
                    brandsSupportedSection.style.display = 'block';
                } else {
                    brandsSupportedSection.style.display = 'none';
                }
                
                // Handle modal hero media - NO CONTROLS IN MODAL EITHER
                const modalHeroMedia = document.getElementById('modal-hero-media');
                if (modalHeroMedia) {
                    modalHeroMedia.innerHTML = '';
                    
                    // Find the clicked card to get its media
                    const clickedCard = document.querySelector(`[data-era="${era}"]`);
                    if (clickedCard) {
                        const media = clickedCard.querySelector('.fs-media');
                        if (media) {
                            const video = media.querySelector('video');
                            const img = media.querySelector('img');
                            
                            if (video) {
                                const modalVideo = video.cloneNode(true);
                                // Remove any controls and ensure premium experience in modal
                                modalVideo.controls = false;
                                modalVideo.autoplay = true;
                                modalVideo.muted = true;
                                modalVideo.loop = true;
                                modalVideo.playsInline = true;
                                modalVideo.disablePictureInPicture = true;
                                modalVideo.setAttribute('controlslist', 'nodownload nofullscreen noremoteplayback');
                                modalVideo.oncontextmenu = function() { return false; };
                                modalVideo.style.pointerEvents = 'none';
                                
                                modalHeroMedia.appendChild(modalVideo);
                                modalVideo.play().catch(() => {});
                            } else if (img) {
                                modalHeroMedia.appendChild(img.cloneNode(true));
                            }
                        }
                    }
                }
                
                modal.classList.add('active');
                document.body.style.overflow = 'hidden';
            }
            
            function closeTimelineModal() {
                if (modal) {
                    modal.classList.remove('active');
                    document.body.style.overflow = '';
                }
            }
            
            // Event delegation for timeline cards (both desktop and mobile)
            document.addEventListener('click', function(e) {
                // Only trigger modal if clicking on card, not video directly
                if (e.target.tagName === 'VIDEO') {
                    e.preventDefault();
                    e.stopPropagation();
                    return false;
                }
                
                const card = e.target.closest('.fs-card');
                if (card && card.dataset.era) {
                    e.preventDefault();
                    e.stopPropagation();
                    openTimelineModal(card.dataset.era);
                }
            });
            
            // Prevent ALL video interactions globally
            document.addEventListener('click', function(e) {
                if (e.target.tagName === 'VIDEO') {
                    e.preventDefault();
                    e.stopPropagation();
                    return false;
                }
            });
            
            document.addEventListener('dblclick', function(e) {
                if (e.target.tagName === 'VIDEO') {
                    e.preventDefault();
                    e.stopPropagation();
                    return false;
                }
            });
            
            // Disable keyboard controls on videos
            document.addEventListener('keydown', function(e) {
                // Disable spacebar, arrow keys, etc. when video is focused
                if (document.activeElement && document.activeElement.tagName === 'VIDEO') {
                    if (e.code === 'Space' || e.code === 'ArrowLeft' || e.code === 'ArrowRight' || 
                        e.code === 'ArrowUp' || e.code === 'ArrowDown' || e.code === 'Enter') {
                        e.preventDefault();
                        return false;
                    }
                }
                
                // ESC to close modal
                if (e.key === 'Escape' && modal && modal.classList.contains('active')) {
                    closeTimelineModal();
                }
            });
            
            // Modal close functionality
            const closeBtn = document.querySelector('.timeline-modal .modal-close');
            const modalOverlay = document.querySelector('.timeline-modal .modal-overlay');
            
            if (closeBtn) {
                closeBtn.addEventListener('click', closeTimelineModal);
            }
            if (modalOverlay) {
                modalOverlay.addEventListener('click', closeTimelineModal);
            }

            // Ensure videos never get focus
            document.addEventListener('focusin', function(e) {
                if (e.target.tagName === 'VIDEO') {
                    e.target.blur();
                }
            });
            
            // Additional insurance: force restart any video that gets paused
            setInterval(() => {
                const videos = document.querySelectorAll('video:not(.modal-content video)');
                videos.forEach(video => {
                    if (video.paused && video.readyState >= 2) {
                        video.currentTime = 0;
                        video.play().catch(() => {});
                    }
                });
            }, 1000);
        });

        // Global protection against video interactions
        document.addEventListener('DOMContentLoaded', function() {
            // Disable drag and drop on videos
            document.addEventListener('dragstart', function(e) {
                if (e.target.tagName === 'VIDEO') {
                    e.preventDefault();
                    return false;
                }
            });
            
            // Prevent text selection on video elements
            document.addEventListener('selectstart', function(e) {
                if (e.target.tagName === 'VIDEO') {
                    e.preventDefault();
                    return false;
                }
            });
        });
    </script>
