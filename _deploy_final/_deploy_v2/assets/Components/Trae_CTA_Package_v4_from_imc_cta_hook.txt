TASK 4 — Detection Script (embed before </body>)
------------------------------------------------
4A) In assets\Contact\contact.html, locate </body>.
4B) **Insert before** </body> the following block:


<!-- ▾ IMC arrival detection + copy swap + attribution ▾ -->
<script>
(function(){
  var params = new URLSearchParams(window.location.search);
  var from = (params.get('from') || '').toLowerCase();

  // 🧾 Attribution
  var lead = document.getElementById('lead_source');
  if (lead) lead.value = from || 'organic';

  if (from === 'imc') {
    // 🎭 Page-level class for variant styling
    document.documentElement.classList.add('from-imcservices');

    // 📝 Copy swap if hooks exist
    var head = document.querySelector('[data-contact-headline]');
    var sub  = document.querySelector('[data-contact-subcopy]');
    if (head) head.textContent = "Let’s scope the right engagement.";
    if (sub)  sub.textContent  = "You came from the IMC Services hub. Tell us where you are in your journey and we’ll map the smartest next step.";

    // 🎯 Optional: preselect Reason dropdown by label
    var reason = document.querySelector('select[name="reason"], #reason');
    if (reason) {
      var preferred = ["IMC Consulting","Integrated Marketing","ScopeIQ follow-up"];
      var found = Array.prototype.find.call(reason.options, function(o){
        return (o.text||'').trim() && preferred.indexOf((o.text||'').trim()) !== -1;
      });
      if (found) reason.value = found.value;
    }

    // 🎯 Ensure anchor lands perfectly with sticky header
    if (window.location.hash === '#cta-hook') {
      var el = document.getElementById('cta-hook');
      if (el) {
        requestAnimationFrame(function(){
          var y = el.getBoundingClientRect().top + window.scrollY - 96; // tweak to header height
          window.scrollTo({ top: y, behavior: 'auto' });
        });
      }
    }
  }
})();
</script>
<!-- ▴ IMC detection end ▴ -->
