SMOOTH SCROLLING FIX - ENGAGEMENT TOOLKIT IMPLEMENTATION GUIDE
================================================================

Target Page: https://luis-gilberto.com/IMCServices/engagement-toolkit/
Implementation Time: 30-60 minutes
Priority: High - User Experience Fix

PROBLEM DESCRIPTION
==================
The current smooth scrolling implementation is too aggressive and intrusive. Users report that "as I scroll once, I end up skipping some of the content" due to forced section-to-section jumping behavior that interferes with natural reading flow.

SOLUTION
========
Replace the existing smooth scrolling system with an elegant, content-aware solution that enhances rather than hijacks the natural scrolling experience. The new system automatically adapts to content density and preserves user control.

=============================

Add this JavaScript code before the closing body tag or in your existing script section:

// Enhanced Smooth Scrolling - Content-Aware & Non-Intrusive
class ElegantSmoothScroll {
    constructor() {
        this.isScrolling = false;
        this.scrollTimer = null;
        this.currentScrollPosition = 0;
        this.targetScrollPosition = 0;
        this.scrollVelocity = 0;
        this.dampingFactor = 0.92;
        this.scrollSensitivity = 1.2;
        this.minScrollDelta = 8;
        
        this.init();
    }
    
    init() {
        this.addScrollProgress();
        this.bindScrollEvents();
        this.currentScrollPosition = window.pageYOffset;
        this.targetScrollPosition = this.currentScrollPosition;
    }
    
    addScrollProgress() {
        const progressBar = document.createElement('div');
        progressBar.className = 'scroll-progress';
        document.body.appendChild(progressBar);
        this.progressBar = progressBar;
    }
    
    bindScrollEvents() {
        window.addEventListener('scroll', this.handleScroll.bind(this), { passive: true });
        window.addEventListener('wheel', this.handleWheel.bind(this), { passive: false });
        window.addEventListener('resize', this.handleResize.bind(this), { passive: true });
        window.addEventListener('touchstart', this.handleTouchStart.bind(this), { passive: true });
        window.addEventListener('touchmove', this.handleTouchMove.bind(this), { passive: false });
        window.addEventListener('touchend', this.handleTouchEnd.bind(this), { passive: true });
    }
    
    handleScroll() {
        this.currentScrollPosition = window.pageYOffset;
        this.updateScrollProgress();
        this.updateVisibleElements();
        
        if (this.scrollTimer) {
            clearTimeout(this.scrollTimer);
        }
        
        document.body.classList.add('is-scrolling');
        
        this.scrollTimer = setTimeout(() => {
            document.body.classList.remove('is-scrolling');
            this.isScrolling = false;
        }, 150);
    }
    
    handleWheel(e) {
        const delta = e.deltaY;
        
        if (Math.abs(delta) < this.minScrollDelta) {
            return;
        }
        
        const scrollMultiplier = this.calculateScrollMultiplier();
        const enhancedDelta = delta * scrollMultiplier * this.scrollSensitivity;
        
        this.scrollVelocity += enhancedDelta * 0.1;
        this.scrollVelocity *= this.dampingFactor;
        
        this.targetScrollPosition += this.scrollVelocity;
        this.targetScrollPosition = Math.max(0, Math.min(
            this.targetScrollPosition,
            document.documentElement.scrollHeight - window.innerHeight
        ));
        
        this.applyParallaxEffects();
    }
    
    calculateScrollMultiplier() {
        const currentSection = this.getCurrentSection();
        
        if (!currentSection) return 1;
        
        const contentElements = currentSection.querySelectorAll('p, li, .card, .feature');
        const contentDensity = contentElements.length;
        
        if (contentDensity > 10) return 0.8;
        if (contentDensity > 5) return 0.9;
        return 1;
    }
    
    getCurrentSection() {
        const sections = document.querySelectorAll('section');
        const scrollPos = window.pageYOffset + (window.innerHeight / 2);
        
        for (let section of sections) {
            const rect = section.getBoundingClientRect();
            const sectionTop = rect.top + window.pageYOffset;
            const sectionBottom = sectionTop + rect.height;
            
            if (scrollPos >= sectionTop && scrollPos <= sectionBottom) {
                return section;
            }
        }
        return null;
    }
    
    applyParallaxEffects() {
        const sections = document.querySelectorAll('section');
        const scrollPos = window.pageYOffset;
        
        sections.forEach((section, index) => {
            const speed = (index % 2 === 0) ? 0.5 : 0.3;
            const yPos = -(scrollPos * speed);
            section.style.setProperty('--parallax-y', `${yPos}px`);
        });
    }
    
    updateScrollProgress() {
        const scrollPercent = this.currentScrollPosition / 
            (document.documentElement.scrollHeight - window.innerHeight);
        
        if (this.progressBar) {
            this.progressBar.style.transform = `scaleX(${Math.min(scrollPercent, 1)})`;
        }
    }
    
    updateVisibleElements() {
        const elementsToAnimate = document.querySelectorAll('.fade-in-on-scroll:not(.visible)');
        
        elementsToAnimate.forEach(element => {
            const rect = element.getBoundingClientRect();
            const threshold = window.innerHeight * 0.8;
            
            if (rect.top < threshold) {
                element.classList.add('visible');
            }
        });
    }
    
    handleTouchStart(e) {
        this.touchStartY = e.touches[0].clientY;
        this.touchStartTime = Date.now();
    }
    
    handleTouchMove(e) {
        if (!this.touchStartY) return;
        
        const touchCurrentY = e.touches[0].clientY;
        const touchDelta = this.touchStartY - touchCurrentY;
        const timeDelta = Date.now() - this.touchStartTime;
        const velocity = touchDelta / timeDelta;
        
        if (Math.abs(velocity) > 0.5) {
            this.scrollVelocity = velocity * 100;
            this.applyMomentumScroll();
        }
    }
    
    handleTouchEnd() {
        this.touchStartY = null;
        this.touchStartTime = null;
    }
    
    applyMomentumScroll() {
        if (Math.abs(this.scrollVelocity) < 0.1) {
            return;
        }
        
        this.targetScrollPosition += this.scrollVelocity;
        this.scrollVelocity *= 0.95;
        
        this.targetScrollPosition = Math.max(0, Math.min(
            this.targetScrollPosition,
            document.documentElement.scrollHeight - window.innerHeight
        ));
        
        requestAnimationFrame(() => this.applyMomentumScroll());
    }
    
    handleResize() {
        this.currentScrollPosition = window.pageYOffset;
        this.targetScrollPosition = this.currentScrollPosition;
        this.updateScrollProgress();
    }
    
    scrollToElement(element, offset = 0) {
        if (!element) return;
        
        const targetPosition = element.getBoundingClientRect().top + 
            window.pageYOffset - offset;
        
        window.scrollTo({
            top: targetPosition,
            behavior: 'smooth'
        });
    }
}

// Initialize elegant smooth scrolling
document.addEventListener('DOMContentLoaded', function() {
    const isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
    
    if (!isMobile) {
        window.elegantScroll = new ElegantSmoothScroll();
    }
    
    const elementsToAnimate = document.querySelectorAll('section > *, .card, .feature-item, .investment-card');
    elementsToAnimate.forEach(element => {
        element.classList.add('fade-in-on-scroll');
    });
});

STEP 5: TEST THE IMPLEMENTATION
------------------------------
After implementing the code, test the following:

FUNCTIONAL TESTING:
- Verify users can scroll normally without forced section jumps
- Confirm all content remains accessible during scroll
- Test that dense content sections scroll slower than light sections
- Verify scroll progress bar appears at top and updates correctly
- Check that content fades in smoothly as it becomes visible
- Ensure mobile devices preserve native momentum scrolling

CROSS-BROWSER TESTING:
- Test in Chrome, Safari, Firefox, and Edge
- Test on desktop, tablet, and mobile devices

TROUBLESHOOTING:
- If scroll progress bar not appearing: Check CSS loading and z-index conflicts
- If fade animations not working: Verify elements have fade-in-on-scroll class
- If performance issues occur: Check for JavaScript conflicts and console errors
- If mobile scrolling feels wrong: Verify mobile detection code is working

EXPECTED RESULT:
Users experience natural, responsive scrolling that feels enhanced but never intrusive. All content remains accessible while adding subtle visual polish.
