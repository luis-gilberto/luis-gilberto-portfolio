        // Enhanced Microsoft-inspired floating elements animation
        function createFloatingElements() {
            const hero = document.querySelector('.about-hero');
            if (!hero) return;
            
            // Remove any existing floating elements first
            hero.querySelectorAll('.floating-element').forEach(el => el.remove());
            
            // Create subtle floating Microsoft-style elements
            const elements = [
                { icon: '⊞', color: 'rgba(0, 120, 212, 0.15)' }, // Windows
                { icon: '◈', color: 'rgba(249, 111, 110, 0.12)' }, // Generic
                { icon: '◊', color: 'rgba(16, 124, 16, 0.1)' }    // Office green
            ];
            
            elements.forEach((item, i) => {
                const element = document.createElement('div');
                element.className = 'floating-element';
                element.innerHTML = item.icon;
                element.style.cssText = `
                    position: absolute;
                    width: 30px;
                    height: 30px;
                    background: ${item.color};
                    border-radius: 6px;
                    top: ${20 + Math.random() * 40}%;
                    left: ${15 + (i * 25) + Math.random() * 15}%;
                    font-size: 18px;
                    display: flex;
                    align-items: center;
                    justify-content: center;
                    color: rgba(255, 255, 255, 0.4);
                    animation: float-subtle ${12 + Math.random() * 8}s ease-in-out infinite;
                    animation-delay: ${i * 2}s;
                    z-index: 2;
                    pointer-events: none;
                    backdrop-filter: blur(10px);
                    border: 1px solid rgba(255, 255, 255, 0.1);
                `;
                hero.appendChild(element);
            });
        }

        // Enhanced typewriter with better Microsoft taglines
        const enhancedTaglines = [
            ' a Microsoft ecosystem strategist',
            ' bringing AI campaigns to life', 
            ' a Venezuelan creative in Seattle',
            ' bridging tech and storytelling',
            ' a former magician turned marketer',
            ' crafting Copilot narratives'
        ];

        // Override existing typewriter if it exists
        let enhancedTypewriterActive = false;
        
        function enhancedTypeWriter() {
            if (enhancedTypewriterActive) return;
            enhancedTypewriterActive = true;
            
            const typewriterElement = document.getElementById('typewriter');
            if (!typewriterElement) return;
            
            let currentIndex = 0;
            let charIndex = 0;
            let isDeleting = false;
            let typeSpeed = 100;
            
            function type() {
                const currentText = enhancedTaglines[currentIndex];
                
                if (isDeleting) {
                    typewriterElement.textContent = currentText.substring(0, charIndex - 1);
                    charIndex--;
                    typeSpeed = 50;
                } else {
                    typewriterElement.textContent = currentText.substring(0, charIndex + 1);
                    charIndex++;
                    typeSpeed = 100;
                }
                
                if (!isDeleting && charIndex === currentText.length) {
                    typeSpeed = 2500; // Longer pause
                    isDeleting = true;
                } else if (isDeleting && charIndex === 0) {
                    isDeleting = false;
                    currentIndex = (currentIndex + 1) % enhancedTaglines.length;
                    typeSpeed = 500;
                }
                
                setTimeout(type, typeSpeed);
            }
            
            setTimeout(type, 1000);
        }

        // Add CSS for floating elements animation
        if (!document.querySelector('#floating-styles')) {
            const style = document.createElement('style');
            style.id = 'floating-styles';
            style.textContent = `
                @keyframes float-subtle {
                    0%, 100% { 
                        transform: translateY(0px) rotate(0deg); 
                        opacity: 0.6; 
                    }
                    25% { 
                        transform: translateY(-15px) rotate(90deg); 
                        opacity: 0.3; 
                    }
                    50% { 
                        transform: translateY(-25px) rotate(180deg);

                    75% {

                .floating-element {
                    transition: all 0.3s ease;
                }
                
                .about-hero:hover .floating-element {
                    animation-duration: 6s !important;
                }
            `;
            document.head.appendChild(style);
        }

        // Initialize enhanced features
        setTimeout(() => {
            createFloatingElements();
            enhancedTypeWriter();
        }, 1200);

        // Reinitialize on window resize
        let resizeTimeout;
        window.addEventListener('resize', () => {
            clearTimeout(resizeTimeout);
            resizeTimeout = setTimeout(() => {
                createFloatingElements();
            }, 250);
        });

        // Add Microsoft-inspired cursor trail effect (subtle)
        let mouseTrail = [];
        const maxTrailLength = 8;
        
        function createTrailElement(x, y) {
            const trail = document.createElement('div');
            trail.style.cssText = `
                position: fixed;
                left: ${x}px;
                top: ${y}px;
                width: 4px;
                height: 4px;
                background: rgba(249, 111, 110, 0.6);
                border-radius: 50%;
                pointer-events: none;
                z-index: 9999;
                transition: all 0.5s ease-out;
            `;
            document.body.appendChild(trail);
            
            setTimeout(() => {
                trail.style.opacity = '0';
                trail.style.transform = 'scale(0)';
                setTimeout(() => trail.remove(), 500);
            }, 50);
        }

        // Only add mouse trail on hero section hover
        document.querySelector('.about-hero')?.addEventListener('mousemove', (e) => {
            if (window.innerWidth > 768) { // Desktop only
                mouseTrail.push({ x: e.clientX, y: e.clientY, time: Date.now() });
                
                if (mouseTrail.length > maxTrailLength) {
                    mouseTrail.shift();
                }
                
                if (Math.random() > 0.7) { // 30% chance to create trail dot
                    createTrailElement(e.clientX - 2, e.clientY - 2);
                }
            }
        });
