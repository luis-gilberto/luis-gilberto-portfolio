```javascript
// Load ScopeIQ submissions from JSON file
async function loadScopeIQSubmissions() {
    try {
        const response = await fetch('/IMCServices/scopeiq-wizard/submissions.json');
        const submissions = await response.json();
        return submissions;
    } catch (error) {
        console.error('Error loading ScopeIQ data:', error);
        return [];
    }
}

// Search clients by name, email, or company
function searchClients(submissions, query) {
    if (!query) return submissions;
    const searchTerm = query.toLowerCase();
    return submissions.filter(client => 
        client.name.toLowerCase().includes(searchTerm) ||
        client.email.toLowerCase().includes(searchTerm) ||
        client.company.toLowerCase().includes(searchTerm)
    );
}

// Populate client context panel with selected client data
function populateClientContext(clientData) {
    // Update client name in dashboard header
    document.getElementById('clientName').textContent = `${clientData.name} - ${clientData.company}`;
    
    // Update client context fields
    document.getElementById('projectType').textContent = mapRecommendationToProject(clientData.recommendation);
    document.getElementById('budgetRange').textContent = getBudgetFromRecommendation(clientData.recommendation);
    document.getElementById('timeline').textContent = getTimelineFromRecommendation(clientData.recommendation);
    document.getElementById('companySize').textContent = clientData.company;
    
    // Store current client data for assessment use
    window.currentClientData = clientData;
}

// Map ScopeIQ recommendations to project types
function mapRecommendationToProject(recommendation) {
    const mapping = {
        'Quick-Start Sprint': 'Go-to-Market Strategy',
        'Fractional Leadership': 'Strategic Consultation', 
        'Launch Campaign': 'Campaign Planning',
        'Growth Partnership': 'Strategic Campaigns',
        'Enterprise Consulting': 'Enterprise Strategy',
        'Founder\'s Ally': 'Strategic Advisory'
    };
    return mapping[recommendation] || 'Strategic Consultation';
}

// Get budget range from recommendation tier
function getBudgetFromRecommendation(recommendation) {
    const budgets = {
        'Quick-Start Sprint': '$4k-$8k',
        'Fractional Leadership': '$3k-$6k/mo',
        'Launch Campaign': '$12k-$25k', 
        'Growth Partnership': '$8k-$15k/mo',
        'Enterprise Consulting': 'Custom',
        'Founder\'s Ally': '$2k-$4k/mo'
    };
    return budgets[recommendation] || 'To be determined';
}

// Get timeline from recommendation
function getTimelineFromRecommendation(recommendation) {
    const timelines = {
        'Quick-Start Sprint': '2-4 weeks',
        'Fractional Leadership': '3-6 months',
        'Launch Campaign': '6-12 weeks',
        'Growth Partnership': '6+ months', 
        'Enterprise Consulting': 'Project-based',
        'Founder\'s Ally': 'Ongoing'
    };
    return timelines[recommendation] || '8-12 weeks';
}
```

C. Add Modal Control Functions:

```javascript
let allScopeIQSubmissions = [];

async function showScopeIQLoader() {
    // Load all submissions
    allScopeIQSubmissions = await loadScopeIQSubmissions();
    
    // Show modal
    document.getElementById('scopeiqModal').classList.remove('hidden');
    
    // Populate client list
    displayClientList(allScopeIQSubmissions);
    
    // Clear search
    document.getElementById('clientSearch').value = '';
}

function closeScopeIQLoader() {
    document.getElementById('scopeiqModal').classList.add('hidden');
}

function displayClientList(submissions) {
    const clientList = document.getElementById('clientList');
    const noResults = document.getElementById('noResults');
    
    if (submissions.length === 0) {
        clientList.innerHTML = '';
        noResults.classList.remove('hidden');
        return;
    }
    
    noResults.classList.add('hidden');
    
    clientList.innerHTML = submissions.map(client => {
        const submissionDate = new Date(client.timestamp).toLocaleDateString();
        return `
            <div class="bg-gray-800 bg-opacity-50 rounded-lg p-6 border border-gray-700 hover:border-coral transition-colors cursor-pointer"
                 onclick="selectClient('${client.email}')">
                <div class="flex items-start justify-between">
                    <div class="flex-1">
                        <div class="flex items-center gap-3 mb-2">
                            <h5 class="font-semibold text-white text-lg">${client.name} - ${client.company}</h5>
                            <span class="status-badge status-complete">Complete</span>
                        </div>
                        <div class="grid md:grid-cols-3 gap-4 text-sm mb-4">
                            <div>
                                <span class="text-gray-400">Recommendation:</span>
                                <div class="text-white font-medium">${client.recommendation}</div>
                            </div>
                            <div>
                                <span class="text-gray-400">Email:</span>
                                <div class="text-white font-medium">${client.email}</div>
                            </div>
                            <div>
                                <span class="text-gray-400">Submitted:</span>
                                <div class="text-white font-medium">${submissionDate}</div>
                            </div>
                        </div>
                    </div>
                    <button class="btn-coral text-white px-4 py-2 rounded-lg text-sm font-medium ml-4">
                        Load Data
                    </button>
                </div>
            </div>
        `;
    }).join('');
}

function filterClients() {
    const query = document.getElementById('clientSearch').value;
    const filteredClients = searchClients(allScopeIQSubmissions, query);
    displayClientList(filteredClients);
}

function selectClient(clientEmail) {
    const selectedClient = allScopeIQSubmissions.find(client => client.email === clientEmail);
    if (selectedClient) {
        populateClientContext(selectedClient);
        closeScopeIQLoader();
        
        // Auto-highlight recommended assessment area
        autoSelectAssessmentArea(selectedClient.recommendation);
        
        // Show success message
        showNotification('Client data loaded successfully!', 'success');
    }
}
```

```javascript
function autoSelectAssessmentArea(recommendation) {
    // Remove any existing highlights
    document.querySelectorAll('.service-card').forEach(card => {
        card.classList.remove('auto-selected');
    });
    
    // Map recommendations to assessment areas
    const assessmentMapping = {
        'Quick-Start Sprint': 'gtm',
        'Launch Campaign': 'campaign', 
        'Growth Partnership': 'campaign',
        'Enterprise Consulting': 'brand',
        'Fractional Leadership': 'brand',
        'Founder\'s Ally': 'creative'
    };
    
    const assessmentType = assessmentMapping[recommendation];
    if (assessmentType) {
        // Find and highlight the corresponding service card
        const serviceCard = document.querySelector(`[onclick="startAssessment('${assessmentType}')"]`);
        if (serviceCard) {
            serviceCard.classList.add('auto-selected');
            serviceCard.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
    }
}
```

```javascript
function showNotification(message, type = 'info') {
    const notification = document.createElement('div');
    notification.className = `fixed top-4 right-4 z-50 px-6 py-4 rounded-lg text-white font-medium transition-all transform translate-x-full`;
    
    if (type === 'success') {
        notification.classList.add('bg-teal');
        notification.innerHTML = `<i class="fas fa-check-circle mr-2"></i>${message}`;
    } else if (type === 'error') {
        notification.classList.add('bg-coral');  
        notification.innerHTML = `<i class="fas fa-exclamation-circle mr-2"></i>${message}`;
    }
    
    document.body.appendChild(notification);
    
    // Animate in
    setTimeout(() => {
        notification.classList.remove('translate-x-full');
    }, 100);
    
    // Animate out and remove
    setTimeout(() => {
        notification.classList.add('translate-x-full');
        setTimeout(() => {
            document.body.removeChild(notification);
        }, 300);
    }, 3000);
}
```
