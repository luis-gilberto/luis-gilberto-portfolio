    <script>
        // Mobile Menu
        const menuBtn = document.getElementById('mobileMenuBtn');
        const drawer = document.getElementById('mobileDrawer');
        const closeDrawer = document.getElementById('drawerClose');
        
        function toggleMenu() {
            drawer.classList.toggle('is-open');
            document.body.style.overflow = drawer.classList.contains('is-open') ? 'hidden' : '';
        }
        if(menuBtn) menuBtn.addEventListener('click', toggleMenu);
        if(closeDrawer) closeDrawer.addEventListener('click', toggleMenu);

        // Video Modal Logic (Optimized)
        function openVideoModal(url, title, desc) {
            const modal = document.getElementById('videoModal');
            const video = document.getElementById('modalVideo');
            
            document.getElementById('modalVideoSource').src = url;
            document.getElementById('modalVideoTitle').textContent = title;
            document.getElementById('modalVideoDescription').textContent = desc;
            
            video.load();
            modal.style.display = 'flex';
            setTimeout(() => { video.play(); }, 100);
        }
        
        document.getElementById('modalClose').addEventListener('click', () => {
            const modal = document.getElementById('videoModal');
            const video = document.getElementById('modalVideo');
            video.pause();
            modal.style.display = 'none';
        });
    </script>
