<script>
  // Gentle pointer-based parallax for hero overlay; respects reduced motion
  (function () {
    const hero = document.querySelector('section.hero#home');
    const overlay = hero ? hero.querySelector('.floating-shapes') : null;
    if (!hero || !overlay) return;

    const prefersReduce = window.matchMedia('(prefers-reduced-motion: reduce)');
    let rafId = null;
    let targetX = 0, targetY = 0;

    const ampX = () => (window.innerWidth <= 768 ? 6 : 10); // px
    const ampY = () => (window.innerWidth <= 768 ? 4 : 8); // px

    function apply() {
      rafId = null;
      overlay.style.setProperty('--parallax-x', targetX.toFixed(2) + 'px');
      overlay.style.setProperty('--parallax-y', targetY.toFixed(2) + 'px');
    }

    function onPointerMove(e) {
      if (prefersReduce.matches) return;
      const rect = hero.getBoundingClientRect();
      const nx = (e.clientX - rect.left) / rect.width - 0.5;
      const ny = (e.clientY - rect.top) / rect.height - 0.5;
      targetX = nx * ampX();
      targetY = ny * ampY();
      if (!rafId) rafId = requestAnimationFrame(apply);
    }

    // Only animate when hero is in view
    const io = new IntersectionObserver((entries) => {
      const entry = entries[0];
      const active = entry && entry.isIntersecting;
      overlay.dataset.parallaxActive = active ? 'true' : 'false';
      if (!active) {
        overlay.style.setProperty('--parallax-x', '0px');
        overlay.style.setProperty('--parallax-y', '0px');
      }
    }, { threshold: 0.2 });
    io.observe(hero);

    hero.addEventListener('pointermove', (e) => {
      if (overlay.dataset.parallaxActive !== 'true') return;
      onPointerMove(e);
    }, { passive: true });

    window.addEventListener('resize', () => {
      overlay.style.setProperty('--parallax-x', '0px');
      overlay.style.setProperty('--parallax-y', '0px');
    });

    prefersReduce.addEventListener('change', () => {
      overlay.style.setProperty('--parallax-x', '0px');
      overlay.style.setProperty('--parallax-y', '0px');
    });
  })();
