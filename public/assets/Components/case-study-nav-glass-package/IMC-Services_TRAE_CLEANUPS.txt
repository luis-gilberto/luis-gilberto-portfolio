(function(){
  var toggle = document.querySelector('[data-action="toggle-pricing"]');
  if (!toggle) return;
  var base = document.querySelectorAll('.base-price');
  var detail = document.querySelectorAll('.detailed-price');
  toggle.addEventListener('click', function(){
    var showDetail = detail[0] && getComputedStyle(detail[0]).display === 'none';

  });
})();

2) Product data array (keep small, easy to read)
Place this near the end of the page (before closing </body>) or in your main JS file:

<script>
(function(){
  const LOGO_BASE = '/assets/logos/'; // adjust to '../assets/logos/' if needed

  const products = [
    { name: 'Microsoft Edge', file: 'Edge.svg', years: '2015 – Present' },
    { name: 'Microsoft Teams', file: 'Teams.svg', years: '2017 – Present' },
    { name: 'Microsoft 365',  file: 'Microsoft365.svg', years: '2017 – Present' },
    { name: 'Copilot',        file: 'Copilot.svg', years: '2023 – Present' },
    { name: 'OneDrive',       file: 'OneDrive.svg', years: '2007 – Present' }
  ];

  // Elements
  const featured = document.getElementById('featuredProduct');
  const prevBtn  = document.getElementById('prevProduct');
  const nextBtn  = document.getElementById('nextProduct');

  if (!featured) return; // safety

  // Render helper
  let idx = 0;
  function render(i){
    const p = products[i];
    if (!p) return;
    featured.innerHTML = `
      <img src="${LOGO_BASE + p.file}" alt="${p.name}" class="spotlight-logo" onerror="this.dataset.missing=1; this.style.opacity=.25;">
      <span class="product-name">${p.name}</span>
      <span class="product-year">${p.years}</span>
    `;
  }

  // Controls
  function clamp(n){ return (n + products.length) % products.length; }
  prevBtn && prevBtn.addEventListener('click', () => { idx = clamp(idx-1); render(idx); });
  nextBtn && nextBtn.addEventListener('click', () => { idx = clamp(idx+1); render(idx); });

  // Auto-rotate (pause on hover)
  let timer = setInterval(() => { idx = clamp(idx+1); render(idx); }, 4000);
  featured.addEventListener('mouseenter', () => clearInterval(timer));
  featured.addEventListener('mouseleave', () => { timer = setInterval(() => { idx = clamp(idx+1); render(idx); }, 4000); });

  // Initial paint
  render(idx);
})();
</script>

E) REMOVE LAST TWO EMPTY SECTIONS (safe automation)

Drop this utility at the bottom of the page (just before </body>). It removes the final two <section> tags ONLY if they contain no visible text or only whitespace:

<script>
(function(){
  function isEffectivelyEmpty(el){
    if (!el) return false;
    const txt = el.textContent.replace(/\s+/g,''); // strip whitespace
    return txt.length === 0;
  }
  const sections = Array.from(document.querySelectorAll('section'));
  let removed = 0;
  for (let i = sections.length - 1; i >= 0 && removed < 2; i--){
    if (isEffectivelyEmpty(sections[i])){
      sections[i].parentNode.removeChild(sections[i]);
      removed++;
    }
  }
})();
</script>

If you prefer a hard delete, simply comment those two <section> blocks in the HTML after you verify they’re truly empty.
