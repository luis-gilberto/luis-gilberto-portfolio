Timeline Fixes Package

Implementation Guide for Luis Gilberto Timeline Issues

Critical Issues Identified

Timeline Endless Cycling Text Cut-off in Cards

Issues Identified

Issue #1: Endless Cycling

The timeline carousel continues scrolling indefinitely instead of stopping at the last visible card.

Root Cause: JavaScript logic doesn't calculate visible cards vs total cards properly.

Issue #2: Text Cut-off

Card descriptions are too long for the fixed-height containers, causing text to be cut off.

Root Cause: Card descriptions exceed the available vertical space in the 320px container.

        // Initialize when DOM loads
        document.addEventListener('DOMContentLoaded', function() {
            // Filmstrip navigation
            let currentIndex = 0;
            const cardsContainer = document.getElementById('cardsContainer');
            const progressBar = document.getElementById('progressBar');
            const prevBtn = document.getElementById('prevBtn');
            const nextBtn = document.getElementById('nextBtn');
            
            // Calculate visible cards and total cards
            const getVisibleCards = () => {
                const trackWidth = document.querySelector('.fs-track').offsetWidth;
                const cardWidth = 280 + 16; // card width + gap
                return Math.floor(trackWidth / cardWidth);
            };
            
            const getTotalCards = () => {
                return document.querySelectorAll('.fs-cards-container .fs-card').length;
            };

            function updateFilmstrip() {
                const visibleCards = getVisibleCards();
                const totalCards = getTotalCards();
                const maxIndex = Math.max(0, totalCards - visibleCards);
                
                // Clamp currentIndex within bounds
                currentIndex = Math.max(0, Math.min(currentIndex, maxIndex));
                
                const cardWidth = 280 + 16; // card width + gap
                const offset = -currentIndex * cardWidth;
                cardsContainer.style.transform = `translateX(${offset}px)`;
                
                // Update progress bar
                const progress = maxIndex > 0 ? ((currentIndex / maxIndex) * 100) : 100;
                progressBar.style.width = `${Math.min(100, Math.max(16.66, progress))}%`;
                
                // Update button states
                prevBtn.disabled = currentIndex <= 0;
                nextBtn.disabled = currentIndex >= maxIndex;
                
                // Update button opacity
                prevBtn.style.opacity = currentIndex <= 0 ? '0.3' : '1';
                nextBtn.style.opacity = currentIndex >= maxIndex ? '0.3' : '1';
            }

            prevBtn.addEventListener('click', () => {
                if (currentIndex > 0) {
                    currentIndex--;
                    updateFilmstrip();
                }
            });

            nextBtn.addEventListener('click', () => {
                const visibleCards = getVisibleCards();
                const totalCards = getTotalCards();
                const maxIndex = Math.max(0, totalCards - visibleCards);
                
                if (currentIndex < maxIndex) {
                    currentIndex++;
                    updateFilmstrip();
                }
            });

            // Initialize filmstrip
            updateFilmstrip();
            
            // Update on window resize
            window.addEventListener('resize', updateFilmstrip);

            // Modal functionality
            const modal = document.getElementById('timeline-modal');
            
            function openTimelineModal(era) {
                if (!modal) return;
                
                const data = timelineData[era] || {};
                
                // Populate modal content
                document.getElementById('modal-title').textContent = data.title || '';
                document.getElementById('modal-quote').textContent = data.quote ? `"${data.quote}"` : '';
                document.getElementById('modal-description').innerHTML = `<p>${data.description || ''}</p>`;
                
                // Populate achievements
                const achievementsList = document.querySelector('#modal-achievements ul');
                achievementsList.innerHTML = '';
                (data.achievements || []).forEach(achievement => {
                    const li = document.createElement('li');
                    li.innerHTML = achievement;
                    achievementsList.appendChild(li);
                });
                
                // Populate brands supported
                const brandsSupportedContainer = document.querySelector('#modal-brands-supported .brands-supported-image');
                const brandsSupportedSection = document.querySelector('#modal-brands-supported');
                if (data.brandsSupported && data.brandsSupported.image) {
                    brandsSupportedContainer.innerHTML = '';
                    const img = document.createElement('img');
                    img.src = data.brandsSupported.image.src;
                    img.alt = data.brandsSupported.image.alt;
                    img.onerror = function() { brandsSupportedSection.style.display = 'none'; };
                    brandsSupportedContainer.appendChild(img);
                    brandsSupportedSection.style.display = 'block';
                } else {
                    brandsSupportedSection.style.display = 'none';
                }
                
                modal.classList.add('active');
                document.body.style.overflow = 'hidden';
            }
            
            function closeTimelineModal() {
                if (modal) {
                    modal.classList.remove('active');
                    document.body.style.overflow = '';
                }
            }
            
            // Event delegation for timeline cards (both desktop and mobile)
            document.addEventListener('click', function(e) {
                const card = e.target.closest('.fs-card');
                if (card && card.dataset.era) {
                    e.preventDefault();
                    e.stopPropagation();
                    openTimelineModal(card.dataset.era);
                }
