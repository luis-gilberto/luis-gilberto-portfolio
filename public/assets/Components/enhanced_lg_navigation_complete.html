## 🐛 TROUBLESHOOTING

### Navigation Doesn't Load
**Issue**: JavaScript errors in console
**Solution**: Check file paths, ensure DOM elements exist

### Logomark Not Visible  
**Issue**: SVG paths or CSS colors incorrect
**Solution**: Verify coral color variable, check SVG viewBox

### Drag Functionality Issues
**Issue**: Navigation doesn't drag smoothly
**Solution**: Check z-index conflicts, verify event listeners

### Cross-Page Active States Wrong
**Issue**: Wrong page highlighted in navigation
**Solution**: Verify URL detection logic in `detectCurrentPage()`

### Mobile Layout Problems
**Issue**: Navigation overlaps content on mobile
**Solution**: Check responsive CSS media queries

⚡ Advanced JavaScript

Replace existing navigation JavaScript with this enhanced version:

/**
 * ========================================
 * ENHANCED LG SMART NAVIGATION SYSTEM
 * Advanced JavaScript with Cross-Page Logic
 * ========================================
 */

class LGSmartNavigation {
  constructor(options = {}) {
    // Configuration
    this.config = {
      autoDetectPage: true,
      savePosition: true,
      keyboardNavigation: true,
      smoothScroll: true,
      animationDuration: 400,
      dragThreshold: 10,
      ...options
    };

    // DOM Elements
    this.elements = {};
    this.currentPage = this.detectCurrentPage();

    // State Management
    this.state = {
      isOpen: false,
      isDragging: false,
      dragOffset: { x: 0, y: 0 },
      activeLink: null,
      keyboardIndex: -1
    };

    // Performance Optimization
    this.throttledScroll = this.throttle(this.updateActiveStates.bind(this), 100);
    this.debouncedResize = this.debounce(this.handleResize.bind(this), 250);

    this.init();
  }

  /**
   * Initialize the navigation system
   */
  init() {
    if (!this.cacheElements()) {
      console.error('LG Navigation: Required elements not found');
      return;
    }

    this.setupEventListeners();
    this.loadSavedPosition();
    this.updateActiveStates();
    this.announceToScreenReader('LG Smart Navigation initialized');

    console.log('✅ LG Smart Navigation System initialized');
  }

  /**
   * Cache DOM elements for performance
   */
  cacheElements() {
    const selectors = {
      wrapper: '.lg-nav-hub-wrapper',
      toggle: '.lg-nav-toggle',
      hub: '.lg-nav-hub',
      close: '.lg-nav-close',
      backdrop: '.lg-backdrop',
      links: '.lg-nav-link',
      sections: '.lg-nav-section'
    };

    for (const [key, selector] of Object.entries(selectors)) {
      if (key === 'links' || key === 'sections') {
        this.elements[key] = document.querySelectorAll(selector);
      } else {
        this.elements[key] = document.querySelector(selector);
      }

      if (!this.elements[key] || (NodeList.prototype.isPrototypeOf(this.elements[key]) && this.elements[key].length === 0)) {
        console.warn(`LG Navigation: ${selector} not found`);
        return false;
      }
    }

    return true;
  }

  /**
   * Setup all event listeners
   */
  setupEventListeners() {
    // Toggle functionality
    this.elements.toggle.addEventListener('click', (e) => {
      e.stopPropagation();
      if (!this.state.isDragging) {
        this.toggleNavigation();
      }
    });

    // Close functionality
    this.elements.close.addEventListener('click', () => this.closeNavigation());
    this.elements.backdrop.addEventListener('click', () => this.closeNavigation());

    // Drag functionality
    this.setupDragListeners();

    // Navigation links
    this.setupNavigationLinks();

    // Keyboard navigation
    if (this.config.keyboardNavigation) {
      this.setupKeyboardNavigation();
    }

    // Window events
    window.addEventListener('scroll', this.throttledScroll);
    window.addEventListener('resize', this.debouncedResize);

    // Page visibility
    document.addEventListener('visibilitychange', () => {
      if (document.hidden && this.state.isOpen) {
        this.closeNavigation();
      }
    });
  }

  /**
   * Setup drag and drop functionality
   */
  setupDragListeners() {
    let startPos = { x: 0, y: 0 };
    let initialPos = { x: 0, y: 0 };

    const handleStart = (e) => {
      // Prevent dragging if clicking inside the navigation panel
      if (e.target.closest('.lg-nav-hub') && this.state.isOpen) return;

      this.state.isDragging = false;

      const clientPos = this.getEventPosition(e);
      startPos = { x: clientPos.x, y: clientPos.y };

      const rect = this.elements.wrapper.getBoundingClientRect();
      initialPos = { x: rect.left, y: rect.top };

      this.state.dragOffset = {
        x: startPos.x - initialPos.x,
        y: startPos.y - initialPos.y
      };

      document.addEventListener('mousemove', handleMove);
      document.addEventListener('mouseup', handleEnd);
      document.addEventListener('touchmove', handleMove, { passive: false });
      document.addEventListener('touchend', handleEnd);

      this.elements.wrapper.style.cursor = 'grabbing';
      e.preventDefault();
    };

    const handleMove = (e) => {
      const clientPos = this.getEventPosition(e);
      const deltaX = Math.abs(clientPos.x - startPos.x);
      const deltaY = Math.abs(clientPos.y - startPos.y);

      if (!this.state.isDragging && (deltaX > this.config.dragThreshold || deltaY > this.config.dragThreshold)) {
        this.state.isDragging = true;
      }

      if (this.state.isDragging) {
        const x = clientPos.x - this.state.dragOffset.x;
        const y = clientPos.y - this.state.dragOffset.y;
        this.setPosition(x, y);
      }

      e.preventDefault();
    };

    const handleEnd = () => {
      if (this.state.isDragging) {
        this.elements.wrapper.style.cursor = 'move';
        if (this.config.savePosition) {
          this.savePosition();
        }

        // Reset dragging state after a short delay
        setTimeout(() => {
          this.state.isDragging = false;
        }, 100);
      }

      document.removeEventListener('mousemove', handleMove);
      document.removeEventListener('mouseup', handleEnd);
      document.removeEventListener('touchmove', handleMove);
      document.removeEventListener('touchend', handleEnd);
    };

    this.elements.wrapper.addEventListener('mousedown', handleStart);
    this.elements.wrapper.addEventListener('touchstart', handleStart, { passive: true });
  }

  /**
   * Setup navigation links with cross-page support
   */
  setupNavigationLinks() {
    this.elements.links.forEach((link, index) => {
      link.addEventListener('click', (e) => {
        this.handleLinkClick(e, link, index);
      });

      // Add keyboard navigation support
      link.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          this.handleLinkClick(e, link, index);
        }
      });
    });
  }

  /**
   * Handle navigation link clicks with intelligent routing
   */
  handleLinkClick(e, link, index) {
    const href = link.getAttribute('href');
    const isExternal = link.classList.contains('lg-external-link');
    const target = link.getAttribute('data-target');
    const page = link.getAttribute('data-page');

    // Update active state
    this.setActiveLink(link);

    if (href.startsWith('#') && target) {
      // Internal anchor navigation
      e.preventDefault();
      const targetElement = document.querySelector(href);

      if (targetElement && this.config.smoothScroll) {
        this.smoothScrollToElement(targetElement);
      }

      this.closeNavigation();
      this.announceToScreenReader(`Navigating to ${target} section`);

    } else if (isExternal && page) {
      // Cross-page navigation
      this.announceToScreenReader(`Navigating to ${page} page`);

      // Store current navigation state for cross-page consistency
      if (this.config.savePosition) {
        localStorage.setItem('lgNavState', JSON.stringify({
          targetPage: page,
          timestamp: Date.now()
        }));
      }

      this.closeNavigation();
      // Allow natural navigation to occur

    } else {
      // Standard navigation
      this.closeNavigation();
    }
  }

  /**
   * Setup keyboard navigation
   */
  setupKeyboardNavigation() {
    document.addEventListener('keydown', (e) => {
      if (!this.state.isOpen) return;

      switch (e.key) {
        case 'Escape':
          e.preventDefault();
          this.closeNavigation();
          break;

        case 'ArrowDown':
          e.preventDefault();
          this.navigateKeyboard(1);
          break;

        case 'ArrowUp':
          e.preventDefault();
          this.navigateKeyboard(-1);
          break;

        case 'Tab':
          // Allow natural tab navigation within the panel
          if (!e.target.closest('.lg-nav-hub')) {
            e.preventDefault();
            this.focusFirstLink();
          }
          break;
      }
    });
  }

  /**
   * Navigate with keyboard
   */
  navigateKeyboard(direction) {
    const links = Array.from(this.elements.links);

    if (this.state.keyboardIndex === -1) {
      this.state.keyboardIndex = direction > 0 ? 0 : links.length - 1;
    } else {
      this.state.keyboardIndex = Math.max(0, Math.min(links.length - 1, this.state.keyboardIndex + direction));
    }

    const targetLink = links[this.state.keyboardIndex];
    if (targetLink) {
      targetLink.focus();
    }
  }

  /**
   * Focus first navigation link
   */
  focusFirstLink() {
    if (this.elements.links.length > 0) {
      this.elements.links[0].focus();
      this.state.keyboardIndex = 0;
    }
  }

  /**
   * Toggle navigation open/close
   */
  toggleNavigation() {
    if (this.state.isOpen) {
      this.closeNavigation();
    } else {
      this.openNavigation();
    }
  }

  /**
   * Open navigation
   */
  openNavigation() {
    this.state.isOpen = true;
    this.elements.hub.classList.add('active');
    this.elements.backdrop.classList.add('active');
    this.elements.hub.setAttribute('aria-hidden', 'false');

    document.body.style.overflow = 'hidden';

    // Focus management for accessibility
    setTimeout(() => {
      this.focusFirstLink();
    }, 100);

    this.announceToScreenReader('Navigation menu opened');
  }

  /**
   * Close navigation
   */
  closeNavigation() {
    this.state.isOpen = false;
    this.state.keyboardIndex = -1;

    this.elements.hub.classList.remove('active');
    this.elements.backdrop.classList.remove('active');
    this.elements.hub.setAttribute('aria-hidden', 'true');

    document.body.style.overflow = '';

    // Return focus to toggle button
    this.elements.toggle.focus();

    this.announceToScreenReader('Navigation menu closed');
  }

  /**
   * Detect current page based on URL
   */
  detectCurrentPage() {
    const path = window.location.pathname;

    if (path.includes('/advisory')) return 'advisory';
    if (path.includes('/scopeiq-wizard')) return 'scopeiq';
    if (path.includes('/IMCServices')) return 'services';

    return 'services'; // Default fallback
  }

  /**
   * Update active states based on current page and scroll position
   */
  updateActiveStates() {
    // Clear all active states
    this.elements.links.forEach(link => link.classList.remove('active'));

    if (this.currentPage === 'services') {
      // For services page, use scroll-based activation
      this.updateScrollBasedActiveStates();
    } else {
      // For other pages, highlight the current page link
      this.updatePageBasedActiveStates();
    }
  }

  /**
   * Update active states based on scroll position (for services page)
   */
  updateScrollBasedActiveStates() {
    const scrollPosition = window.scrollY + 100;

    this.elements.links.forEach(link => {
      const href = link.getAttribute('href');
      if (href && href.startsWith('#')) {
        const target = document.querySelector(href);
        if (target) {
          const targetTop = target.offsetTop;
          const targetBottom = targetTop + target.offsetHeight;

          if (scrollPosition >= targetTop && scrollPosition < targetBottom) {
            this.setActiveLink(link);
          }
        }
      }
    });
  }

  /**
   * Update active states based on current page
   */
  updatePageBasedActiveStates() {
    this.elements.links.forEach(link => {
      const page = link.getAttribute('data-page');
      if (page === this.currentPage) {
        this.setActiveLink(link);
      }
    });
  }

  /**
   * Set active link
   */
  setActiveLink(link) {
    if (this.state.activeLink) {
      this.state.activeLink.classList.remove('active');
    }

    link.classList.add('active');
    this.state.activeLink = link;
  }

  /**
   * Smooth scroll to element
   */
  smoothScrollToElement(element) {
    const targetPosition = element.offsetTop - 80; // Account for fixed headers

    window.scrollTo({
      top: targetPosition,
      behavior: 'smooth'
    });
  }

  /**
   * Set navigation position with bounds checking
   */
  setPosition(x, y) {
    const maxX = window.innerWidth - this.elements.wrapper.offsetWidth;
    const maxY = window.innerHeight - this.elements.wrapper.offsetHeight;

    x = Math.max(0, Math.min(x, maxX));
    y = Math.max(0, Math.min(y, maxY));

    this.elements.wrapper.style.left = x + 'px';
    this.elements.wrapper.style.top = y + 'px';
  }

  /**
   * Save current position to localStorage
   */
  savePosition() {
    if (!this.config.savePosition) return;

    const rect = this.elements.wrapper.getBoundingClientRect();
    const position = {
      x: rect.left,
      y: rect.top,
      timestamp: Date.now()
    };

    localStorage.setItem('lgNavPosition', JSON.stringify(position));
  }

  /**
   * Load saved position from localStorage
   */
  loadSavedPosition() {
    if (!this.config.savePosition) return;

    const saved = localStorage.getItem('lgNavPosition');
    if (saved) {
      try {
        const position = JSON.parse(saved);

        // Only use saved position if it's recent (within 24 hours)
        if (Date.now() - position.timestamp < 24 * 60 * 60 * 1000) {
          this.setPosition(position.x, position.y);
        }
      } catch (e) {
        console.warn('LG Navigation: Failed to parse saved position');
      }
    }
  }

  /**
   * Handle window resize
   */
  handleResize() {
    // Ensure navigation stays within viewport bounds
    const rect = this.elements.wrapper.getBoundingClientRect();
    this.setPosition(rect.left, rect.top);

    // Close navigation on mobile resize
    if (window.innerWidth <= 768 && this.state.isOpen) {
      this.closeNavigation();
    }
  }

  /**
   * Get event position (mouse or touch)
   */
  getEventPosition(e) {
    if (e.type.startsWith('touch')) {
      const touch = e.touches[0] || e.changedTouches[0];
      return { x: touch.clientX, y: touch.clientY };
    }
    return { x: e.clientX, y: e.clientY };
  }

  /**
   * Announce to screen readers
   */
  announceToScreenReader(message) {
    const announcer = document.getElementById('lg-nav-instructions');
    if (announcer) {
      announcer.textContent = message;
    }
  }

  /**
   * Throttle function for performance
   */
  throttle(func, limit) {
    let inThrottle;
    return function() {
      const args = arguments;
      const context = this;
      if (!inThrottle) {
        func.apply(context, args);
        inThrottle = true;
        setTimeout(() => inThrottle = false, limit);
      }
    };
  }

  /**
   * Debounce function for performance
   */
  debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
      const later = () => {
        clearTimeout(timeout);
        func(...args);
      };
      clearTimeout(timeout);
      timeout = setTimeout(later, wait);
    };
  }

  /**
   * Public API for external control
   */
  getAPI() {
    return {
      open: () => this.openNavigation(),
      close: () => this.closeNavigation(),
      toggle: () => this.toggleNavigation(),
      setPosition: (x, y) => this.setPosition(x, y),
      getCurrentPage: () => this.currentPage,
      isOpen: () => this.state.isOpen
    };
  }

  /**
   * Cleanup method
   */
  destroy() {
    // Remove all event listeners
    window.removeEventListener('scroll', this.throttledScroll);
    window.removeEventListener('resize', this.debouncedResize);

    // Reset styles
    document.body.style.overflow = '';

    console.log('🧹 LG Smart Navigation destroyed');
  }
}

/**
 * Auto-initialization with DOM ready check
 */
function initLGNavigation(options = {}) {
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', () => {
      window.lgNavigation = new LGSmartNavigation(options);
    });
  } else {
    window.lgNavigation = new LGSmartNavigation(options);
  }
}

/**
 * Initialize with default settings
 */
initLGNavigation({
  autoDetectPage: true,
  savePosition: true,
  keyboardNavigation: true,
  smoothScroll: true
});

/**
 * Export for module systems
 */
if (typeof module !== 'undefined' && module.exports) {
  module.exports = LGSmartNavigation;
}

// Make API available globally for debugging
window.LGSmartNavigation = LGSmartNavigation;
