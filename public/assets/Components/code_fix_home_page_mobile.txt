document.addEventListener('DOMContentLoaded', function() {
    const nav = document.querySelector('.enhanced-nav');
    const mobileMenuBtn = document.getElementById('mobile-menu-btn');
    const mobileMenuOverlay = document.getElementById('mobileMenuOverlay'); // This should match your HTML
    const mobileMenuClose = document.getElementById('mobileMenuClose'); // This should match your HTML

    // Debug: Check if elements exist
    console.log('Navigation elements found:', {
        nav: !!nav,
        mobileMenuBtn: !!mobileMenuBtn,
        mobileMenuOverlay: !!mobileMenuOverlay,
        mobileMenuClose: !!mobileMenuClose
    });

    // Scroll effect for navigation
    window.addEventListener('scroll', function() {
        if (window.scrollY > 50) {
            if (nav) nav.classList.add('scrolled');
        } else {
            if (nav) nav.classList.remove('scrolled');
        }
    });

    // Mobile menu functionality
    function openMobileMenu() {
        if (mobileMenuOverlay) {
            mobileMenuOverlay.classList.add('active');
            document.body.style.overflow = 'hidden';
            console.log('Mobile menu opened');
        } else {
            console.error('Mobile menu overlay not found');
        }
    }

    function closeMobileMenu() {
        if (mobileMenuOverlay) {
            mobileMenuOverlay.classList.remove('active');
            document.body.style.overflow = '';
            console.log('Mobile menu closed');
        }
    }

    // Event listeners with improved error handling
    if (mobileMenuBtn) {
        mobileMenuBtn.addEventListener('click', function(e) {
            e.preventDefault();
            e.stopPropagation();
            console.log('Mobile menu button clicked');
            openMobileMenu();
        });
        
        // Also add touchstart for better mobile support
        mobileMenuBtn.addEventListener('touchstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            console.log('Mobile menu button touched');
            openMobileMenu();
        }, { passive: false });
    } else {
        console.error('Mobile menu button (#mobile-menu-btn) not found');
    }

    if (mobileMenuClose) {
        mobileMenuClose.addEventListener('click', function(e) {
            e.preventDefault();
            e.stopPropagation();
            console.log('Close button clicked');
            closeMobileMenu();
        });
        
        mobileMenuClose.addEventListener('touchstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            closeMobileMenu();
        }, { passive: false });
    } else {
        console.error('Mobile menu close button (#mobileMenuClose) not found');
    }

    // Close menu when clicking overlay background
    if (mobileMenuOverlay) {
        mobileMenuOverlay.addEventListener('click', function(e) {
            if (e.target === mobileMenuOverlay) {
                console.log('Overlay background clicked');
                closeMobileMenu();
            }
        });
        
        // Add touch support for overlay
        mobileMenuOverlay.addEventListener('touchstart', function(e) {
            if (e.target === mobileMenuOverlay) {
                e.preventDefault();
                closeMobileMenu();
            }
        }, { passive: false });
    }

    // Close mobile menu when clicking on a link
    const mobileMenuLinks = document.querySelectorAll('.mobile-menu-link');
    mobileMenuLinks.forEach(link => {
        link.addEventListener('click', function() {
            console.log('Menu link clicked, closing menu');
            closeMobileMenu();
        });
    });

    // Close menu on escape key
    document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' && mobileMenuOverlay && mobileMenuOverlay.classList.contains('active')) {
            console.log('Escape key pressed, closing menu');
            closeMobileMenu();
        }
    });

    // Additional debugging for mobile button visibility and clickability
    if (mobileMenuBtn) {
        const computedStyle = window.getComputedStyle(mobileMenuBtn);
        console.log('Mobile menu button styles:', {
            display: computedStyle.display,
            visibility: computedStyle.visibility,
            opacity: computedStyle.opacity,
            pointerEvents: computedStyle.pointerEvents,
            zIndex: computedStyle.zIndex
        });
        
        // Check if button is actually clickable
        const rect = mobileMenuBtn.getBoundingClientRect();
        console.log('Mobile menu button position:', {
            top: rect.top,
            left: rect.left,
            width: rect.width,
            height: rect.height,
            visible: rect.width > 0 && rect.height > 0
        });
    }
});
