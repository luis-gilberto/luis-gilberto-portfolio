IMPLEMENTATION (Updated)
===========================================

For dual-video theme switching, use this HTML:

<div class="insights-video-title">
    <video autoplay loop muted playsinline class="insights-video">
        <!-- Light mode sources (default) -->
        <source src="/assets/insights-video-light.webm" type="video/webm" class="light-source">
        <source src="/assets/insights-video-light.mp4" type="video/mp4" class="light-source">
        
        <!-- Dark mode sources (hidden by default) -->
        <source src="/assets/insights-video-dark-perfect.webm" type="video/webm" class="dark-source" style="display: none;">
        <source src="/assets/insights-video-dark-perfect.mp4" type="video/mp4" class="dark-source" style="display: none;">
    </video>
</div>

JavaScript for theme switching:
function updateVideoTheme() {
    const video = document.querySelector('.insights-video');
    const theme = document.documentElement.getAttribute('data-theme');
    
    if (theme === 'dark') {
        video.innerHTML = `
            <source src="/assets/insights-video-dark-perfect.webm" type="video/webm">
            <source src="/assets/insights-video-dark-perfect.mp4" type="video/mp4">
        `;
    } else {
        video.innerHTML = `
            <source src="/assets/insights-video-light.webm" type="video/webm">
            <source src="/assets/insights-video-light.mp4" type="video/mp4">
        `;
    }
    
    video.load();
    video.play().catch(() => {});
}
