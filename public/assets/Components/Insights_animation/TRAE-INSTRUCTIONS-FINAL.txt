STEP 2: ADD JAVASCRIPT THEME SWITCHER
-------------------------------------
Add this JavaScript to your page (after the video element):

<script>
// Theme-aware video source switching
function updateInsightsVideo() {
    const video = document.querySelector('.insights-video');
    if (!video) return;
    
    const theme = document.documentElement.getAttribute('data-theme') || 'light';
    
    if (theme === 'dark') {
        video.innerHTML = `
            <source src="/assets/insights-video-dark-perfect.webm" type="video/webm">
            <source src="/assets/insights-video-dark-perfect.mp4" type="video/mp4">
        `;
    } else {
        video.innerHTML = `
            <source src="/assets/insights-video-light.webm" type="video/webm">
            <source src="/assets/insights-video-light.mp4" type="video/mp4">
        `;
    }
    
    video.load();
    video.play().catch(() => {}); // Handle autoplay restrictions
}

// Run on page load
document.addEventListener('DOMContentLoaded', updateInsightsVideo);

// IMPORTANT: Call this function when your theme toggle button is clicked
// Add this line to your existing theme toggle handler:

</script>
