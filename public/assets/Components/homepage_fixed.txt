    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const tiles = document.querySelectorAll('.tour-card');
            const tilesContainer = document.getElementById('tilesContainer');
            const heroSection = document.getElementById('heroSection');

            // Phase 1: Add spread classes (0-0.1s)
            setTimeout(() => {
                tiles.forEach((tile, index) => {
                    tile.classList.add('tile-initial');
                    tile.classList.add(`tile-${index + 1}-initial`);
                    tile.style.opacity = '0.6';
                });
            }, 100);

            // Phase 2: Remove spread, let tiles gather (0.1-3s)
            setTimeout(() => {
                tiles.forEach((tile, index) => {
                    tile.classList.remove('tile-initial');
                    tile.classList.remove(`tile-${index + 1}-initial`);
                    tile.classList.add('gathering');
                });
            }, 3000);

            // Phase 3: Fade out tiles (4-4.5s)
            setTimeout(() => {
                tilesContainer.classList.add('tiles-fadeout');
            }, 4000);

            // Phase 4: Reveal hero (4.5s+)
            setTimeout(() => {
                heroSection.classList.add('hero-visible');
            }, 4500);
        });

        // Mobile Navigation Functionality
        const mobileNavButton = document.getElementById('mobile-nav-button');
        const mobileMenuOverlay = document.getElementById('mobile-menu-overlay');
        const mobileMenuClose = document.getElementById('mobile-menu-close');

        function openMobileMenu() {
            mobileMenuOverlay.classList.add('active');
            document.body.style.overflow = 'hidden';
        }

        function closeMobileMenu() {
            mobileMenuOverlay.classList.remove('active');
            document.body.style.overflow = '';
        }

        if (mobileNavButton) {
            mobileNavButton.addEventListener('click', openMobileMenu);
        }

        if (mobileMenuClose) {
            mobileMenuClose.addEventListener('click', closeMobileMenu);
        }

        // Close mobile menu when clicking outside
        mobileMenuOverlay.addEventListener('click', function(e) {
            if (e.target === mobileMenuOverlay) {
                closeMobileMenu();
            }
        });

        // Close mobile menu on escape key
        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape' && mobileMenuOverlay.classList.contains('active')) {
                closeMobileMenu();
            }
        });
    </script>
