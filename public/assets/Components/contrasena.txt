STEP C: Replace the admin mode JavaScript code you showed me with this:

// ===================================
// Admin Mode Toggle with Password Modal
// ===================================
let adminModeActive = false;
const ADMIN_PASSWORD = 'LG100';
const adminToggleBtn = document.getElementById('admin-toggle');
const adminContents = document.querySelectorAll('.admin-content');
const passwordModal = document.getElementById('passwordModal');
const adminPasswordForm = document.getElementById('adminPasswordForm');
const adminPasswordInput = document.getElementById('adminPasswordInput');
const modalCancel = document.getElementById('modalCancel');
const modalTogglePassword = document.getElementById('modalTogglePassword');
const modalErrorMessage = document.getElementById('modalErrorMessage');

adminToggleBtn.addEventListener('click', () => {
    if (!adminModeActive) {
        // Show password modal
        passwordModal.style.display = 'flex';
        setTimeout(() => adminPasswordInput.focus(), 100);
    } else {
        // Deactivate admin mode
        adminModeActive = false;
        adminToggleBtn.classList.remove('active');
        adminContents.forEach(content => content.style.display = 'none');
        sessionStorage.removeItem('adminMode');
        console.log('%c🔒 Admin Mode Deactivated', 'font-size: 16px; color: #999;');
    }
});

// Handle password form submission
adminPasswordForm.addEventListener('submit', (e) => {
    e.preventDefault();
    modalErrorMessage.style.display = 'none';
    
    if (adminPasswordInput.value === ADMIN_PASSWORD) {
        // Correct password
        adminModeActive = true;
        adminToggleBtn.classList.add('active');
        adminContents.forEach(content => content.style.display = 'block');
        sessionStorage.setItem('adminMode', 'true');
        passwordModal.style.display = 'none';
        adminPasswordInput.value = '';
        console.log('%c🔓 Admin Mode Activated', 'font-size: 16px; font-weight: bold; color: #F96F6E;');
    } else {
        // Wrong password
        modalErrorMessage.style.display = 'block';
        adminPasswordInput.value = '';
        adminPasswordInput.focus();
    }
});

// Cancel button
modalCancel.addEventListener('click', () => {
    passwordModal.style.display = 'none';
    adminPasswordInput.value = '';
    modalErrorMessage.style.display = 'none';
});

// Close on overlay click
passwordModal.addEventListener('click', (e) => {
    if (e.target === passwordModal || e.target.classList.contains('password-modal-overlay')) {
        passwordModal.style.display = 'none';
        adminPasswordInput.value = '';
        modalErrorMessage.style.display = 'none';
    }
});

// Toggle password visibility
modalTogglePassword.addEventListener('click', () => {
    adminPasswordInput.type = adminPasswordInput.type === 'password' ? 'text' : 'password';
});

// Check if admin mode was previously activated
if (sessionStorage.getItem('adminMode') === 'true') {
    adminModeActive = true;
    adminToggleBtn.classList.add('active');
    adminContents.forEach(content => content.style.display = 'block');
}
