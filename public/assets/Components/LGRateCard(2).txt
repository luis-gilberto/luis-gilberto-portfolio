  <script>
    // Utility functions
    const qs = new URLSearchParams(location.search);
    const $ = (s, p=document) => p.querySelector(s);
    const $$ = (s, p=document) => Array.from(p.querySelectorAll(s));

    // Apply incoming query params
    (function initFromQuery(){
      const show = (qs.get('show')||'intro,offer,trust,models,footer')
        .split(',').map(s=>s.trim().toLowerCase());
      const pkgs = (qs.get('packages')||'quick,fractional,launch,ally')
        .split(',').map(s=>s.trim());

      // Sections map
      const map = {
        intro:'sec-intro', offer:'sec-offer', trust:'sec-trust',
        models:'sec-models', footer:'sec-footer'
      };
      
      // Reflect in UI
      Object.entries(map).forEach(([k,id])=>{
        const el = document.getElementById(id);
        const on = show.includes(k);
        el.style.display = on ? '' : 'none';
        const chk = document.querySelector(`[data-toggle="${id}"]`);
        if (chk) chk.checked = on;
      });

      // Packages
      $$('.pkg').forEach(row=>{
        const id = row.dataset.id;
        const on = pkgs.includes(id);
        row.style.display = on ? 'block' : 'none';
        const chk = document.querySelector(`[data-package="${id}"]`);
        if (chk) chk.checked = on;
      });
    })();

    // Live toggles
    $$('#configBar [data-toggle]').forEach(chk=>{
      chk.addEventListener('change', e=>{
        const id = e.target.getAttribute('data-toggle');
        const el = document.getElementById(id);
        if (!el) return;
        el.style.display = e.target.checked ? '' : 'none';
      });
    });

    $$('#configBar [data-package]').forEach(chk=>{
      chk.addEventListener('change', e=>{
        const id = e.target.getAttribute('data-package');
        const row = document.querySelector(`.pkg[data-id="${id}"]`);
        if (row) row.style.display = e.target.checked ? 'block' : 'none';
      });
    });

    // Copy share link
    $('#copyLink').addEventListener('click', ()=>{
      const sections = [];
      const map = {
        'sec-intro':'intro','sec-offer':'offer','sec-trust':'trust',
        'sec-models':'models','sec-footer':'footer'
      };
      Object.entries(map).forEach(([id,key])=>{
        if (document.getElementById(id).style.display !== 'none') sections.push(key);
      });
      const pkgs = $$('.pkg').filter(r=>r.style.display!=='none').map(r=>r.dataset.id);
      const url = new URL(location.href);
      url.searchParams.set('show', sections.join(','));
      url.searchParams.set('packages', pkgs.join(','));
