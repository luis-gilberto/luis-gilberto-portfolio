STEP 4: VERIFY THE JAVASCRIPT
================================================================================

The existing JavaScript at the bottom of the file should remain UNCHANGED:

<script>
  // Pricing password reveal
  (function(){
    const btn = document.getElementById('revealPriceBtn');
    const pwd = document.getElementById('pricePassword');
    const status = document.getElementById('priceStatus');
    function reveal() {
      const val = (pwd && pwd.value ? pwd.value.trim() : '');
      if (val === 'LG100') {
        document.querySelectorAll('.tier-price').forEach(el => el.classList.add('revealed'));
        if (status) { status.textContent = 'Pricing revealed'; status.classList.remove('error'); status.classList.add('ok'); }
      } else {
        if (status) { status.textContent = 'Incorrect password'; status.classList.remove('ok'); status.classList.add('error'); }
      }
    }
    if (btn) btn.addEventListener('click', reveal);
    if (pwd) pwd.addEventListener('keyup', function(e){ if (e.key === 'Enter') reveal(); });
  })();
</script>

✓ No changes needed to this script.
