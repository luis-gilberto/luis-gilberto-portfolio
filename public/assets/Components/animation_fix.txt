---REPLACE WITH THIS CODE (NEW):---
onComplete: function() {
  const loaderContainer = document.querySelector('.hub-loading-container');
  
  // Hold logo for 2 seconds, then trigger fade out
  setTimeout(() => {
    // Add class to trigger logo fade-out via CSS
    if (loaderContainer) {
      loaderContainer.classList.add('is-complete');
    }
    
    // Start the frame reveal 500ms after logo begins fading
    setTimeout(() => {
      revealSystem.style.opacity = '1';

      setTimeout(() => {
        revealSystem.classList.add('is-revealing-frame');
      }, 0);

      setTimeout(() => {
        revealSystem.classList.add('is-revealing-shapes');
      }, 200);
      
      setTimeout(() => {
        showContentAndFinalShapes();
      }, 1500);
      
      setTimeout(() => {
        revealSystem.classList.add('is-fading-out');
      }, 2000);

      setTimeout(() => {
        revealSystem.style.display = 'none';
      }, 3000);
      
    }, 500); // 500ms delay for logo fade + frame reveal overlap
    
  }, 2000); // 2000ms = 2 second hold on logo after video ends
}
---END NEW CODE---
