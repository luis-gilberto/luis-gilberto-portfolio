<script>
(function() {
    const html = document.documentElement;

    // Theme
    const savedTheme = localStorage.getItem('theme') || 'light';
    html.setAttribute('data-theme', savedTheme);
    updateThemeIcons(savedTheme);

    function toggleTheme() {
        const current = html.getAttribute('data-theme');
        const next = current === 'dark' ? 'light' : 'dark';
        html.setAttribute('data-theme', next);
        localStorage.setItem('theme', next);
        updateThemeIcons(next);
    }
    function updateThemeIcons(theme) {
        document.querySelectorAll('#themeToggle i, #drawerThemeToggle i').forEach(icon => {
            icon.className = theme === 'dark' ? 'fa-regular fa-sun' : 'fa-regular fa-moon';
        });
    }
    document.getElementById('themeToggle')?.addEventListener('click', toggleTheme);
    document.getElementById('drawerThemeToggle')?.addEventListener('click', toggleTheme);

    // Drawer
    const mobileToggle = document.getElementById('mobileMenuToggle');
    const mobileDrawer = document.getElementById('mobileDrawer');
    const drawerOverlay = document.getElementById('drawerOverlay');
    const drawerClose = document.getElementById('drawerClose');

    function openDrawer() {
        mobileDrawer.classList.add('active');
        drawerOverlay.classList.add('active');
        document.body.style.overflow = 'hidden';
        mobileToggle.querySelector('i').className = 'fa-solid fa-xmark';
    }
    function closeDrawer() {
        mobileDrawer.classList.remove('active');
        drawerOverlay.classList.remove('active');
        document.body.style.overflow = '';
        mobileToggle.querySelector('i').className = 'fa-solid fa-bars';
    }
    mobileToggle?.addEventListener('click', () => {
        mobileDrawer.classList.contains('active') ? closeDrawer() : openDrawer();
    });
    drawerClose?.addEventListener('click', closeDrawer);
    drawerOverlay?.addEventListener('click', closeDrawer);
    document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && mobileDrawer.classList.contains('active')) closeDrawer();
    });

    // Footer Year
    const yearEl = document.getElementById('currentYear');
    if (yearEl) yearEl.textContent = new Date().getFullYear();

    // Price Lock
    const revealBtn = document.getElementById('revealPriceBtn');
    const passwordInput = document.getElementById('pricePassword');
    const statusDiv = document.getElementById('priceStatus');
    const prices = document.querySelectorAll('.tier-price');
    if (revealBtn) {
        revealBtn.addEventListener('click', () => {
            if (passwordInput.value === 'LG100') {
                prices.forEach(p => p.classList.add('revealed'));
                statusDiv.style.display = 'block';
                statusDiv.textContent = 'Pricing Revealed';
                statusDiv.style.color = 'var(--teal)';
                document.getElementById('priceLock').style.borderColor = 'var(--teal)';
            } else {
                statusDiv.style.display = 'block';
                statusDiv.textContent = 'Incorrect Password';
                statusDiv.style.color = 'var(--coral)';
                document.getElementById('priceLock').style.borderColor = 'var(--coral)';
            }
        });
    }

    // Logo Rotator
    const logos = [
        { name: 'Microsoft Copilot', path: '/TheHub/assets/images/Copilot.png' },
        { name: 'Copilot in Edge', path: '/TheHub/assets/images/Copilot_in_Edge.png' },
        { name: 'Family Safety', path: '/TheHub/assets/images/FamilySafety.png' },
        { name: 'M365 Free Apps', path: '/TheHub/assets/images/M365_Free_apps.png' },
        { name: 'Microsoft Band', path: '/TheHub/assets/images/Microsoft_Band.png' },
        { name: 'Microsoft 365', path: '/TheHub/assets/images/Microsoft365.png' },
        { name: 'Office 365', path: '/TheHub/assets/images/O365.png' },
        { name: 'Office 365 Legacy', path: '/TheHub/assets/images/Office365.svg' },
        { name: 'Microsoft Edge', path: '/TheHub/assets/images/Edge.png' },
        { name: 'Microsoft Teams', path: '/TheHub/assets/images/Teams.png' }
    ];
    const logoContainer = document.querySelector('.logo-display');
    if (logoContainer) {
        let currentIdx = 0;
        logos.forEach((logo, index) => {
            const item = document.createElement('div');
            item.className = `logo-item ${index === 0 ? 'active' : ''}`;
            item.innerHTML = `<img src="${logo.path}" alt="${logo.name}">`;
            logoContainer.appendChild(item);
        });
        setInterval(() => {
            const items = document.querySelectorAll('.logo-item');
            items[currentIdx].classList.remove('active');
            currentIdx = (currentIdx + 1) % items.length;
            items[currentIdx].classList.add('active');
        }, 3000);
    }
})();
