    <script>
        // MOBILE DRAWER
        const overlay = document.getElementById('mobileMenuOverlay');
        const openBtn = document.getElementById('mobileMenuBtn');
        const closeBtn = document.getElementById('mobileMenuClose');
        
        if (openBtn) {
            openBtn.addEventListener('click', () => {
                overlay.classList.add('active');
                document.body.style.overflow = 'hidden';
            });
        }
        if (closeBtn) {
            closeBtn.addEventListener('click', () => {
                overlay.classList.remove('active');
                document.body.style.overflow = '';
            });
        }

        // ARTIFACT CONFIGURATOR
        const swatches = document.querySelectorAll('.palette-option');
        const previewSurface = document.getElementById('resumePreviewList');
        const downloadBtn = document.getElementById('downloadBtn');

        swatches.forEach(swatch => {
            swatch.addEventListener('click', () => {
                const style = swatch.dataset.style;
                const pdf = swatch.dataset.pdf;
                
                swatches.forEach(s => s.classList.remove('active'));
                swatch.classList.add('active');
                
                previewSurface.setAttribute('data-preview-style', style);
                downloadBtn.href = `assets/Components/THE_FOUR/${pdf}`;
            });
        });

        // THEME
        document.getElementById('themeToggle').addEventListener('click', () => {
            const html = document.documentElement;
            const newTheme = html.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
            html.setAttribute('data-theme', newTheme);
            localStorage.setItem('theme', newTheme);
        });
    </script>
