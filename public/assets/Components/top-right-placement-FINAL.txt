Replace your entire styling section with this:

          // Mysterious, subtle styling - positioned TOP-RIGHT
          Object.assign(learnMoreCTA.style, {
            position: 'fixed',
            right: '2rem',
            top: '2rem',
            zIndex: '10001',
            display: 'inline-flex',
            alignItems: 'center',
            gap: '0.5rem',
            padding: '0.75rem 1.25rem',
            background: 'rgba(46, 211, 198, 0.08)',
            backdropFilter: 'blur(10px)',
            WebkitBackdropFilter: 'blur(10px)',
            border: '1px solid rgba(46, 211, 198, 0.2)',
            borderRadius: '8px',
            color: 'rgba(46, 211, 198, 0.7)',
            textDecoration: 'none',
            fontFamily: 'Inter, sans-serif',
            fontWeight: '500',
            fontSize: '0.875rem',
            letterSpacing: '0.02em',
            opacity: '0',
            transform: 'translateY(-10px)',  // Comes from above
            transition: 'all 0.4s cubic-bezier(0.16, 1, 0.3, 1)',
            pointerEvents: 'none',
            boxShadow: '0 0 0 rgba(46, 211, 198, 0)',
            textShadow: '0 0 0 rgba(46, 211, 198, 0)'
          });
          
          // Mobile responsive
          if (window.innerWidth <= 768) {
            learnMoreCTA.style.right = '1rem';
            learnMoreCTA.style.top = '1rem';
            learnMoreCTA.style.padding = '0.625rem 1rem';
            learnMoreCTA.style.fontSize = '0.8125rem';
          }
          
          // Hover: intensify the mystery
          learnMoreCTA.addEventListener('mouseenter', function() {
            this.style.background = 'rgba(46, 211, 198, 0.18)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.5)';
            this.style.color = 'rgba(46, 211, 198, 1)';
            this.style.transform = 'translateY(0) scale(1.02)';
            this.style.boxShadow = '0 0 30px rgba(46, 211, 198, 0.3), 0 0 60px rgba(46, 211, 198, 0.1)';
            this.style.textShadow = '0 0 20px rgba(46, 211, 198, 0.5)';
            
            const svg = this.querySelector('.deeper-arrow');
            if (svg) svg.style.transform = 'translateX(4px)';
          });
          
          learnMoreCTA.addEventListener('mouseleave', function() {
            this.style.background = 'rgba(46, 211, 198, 0.08)';
            this.style.borderColor = 'rgba(46, 211, 198, 0.2)';
            this.style.color = 'rgba(46, 211, 198, 0.7)';
            this.style.transform = 'translateY(0) scale(1)';
            this.style.boxShadow = '0 0 20px rgba(46, 211, 198, 0.15)';
            this.style.textShadow = '0 0 10px rgba(46, 211, 198, 0.3)';
            
            const svg = this.querySelector('.deeper-arrow');
            if (svg) svg.style.transform = 'translateX(0)';
          });
          
          // Style the arrow
          const svg = learnMoreCTA.querySelector('.deeper-arrow');
          if (svg) {
            svg.style.transition = 'transform 0.3s ease';
            svg.style.opacity = '0.7';
          }
          
          // Track navigation
          learnMoreCTA.addEventListener('click', function(e) {
            sessionStorage.setItem('hubNavigated', 'true');
          });
          
          // Add to container
          hubContainer.appendChild(learnMoreCTA);
          
          // TEASING REVEAL: Fade in slowly from above
          setTimeout(() => {
            learnMoreCTA.style.opacity = '1';
            learnMoreCTA.style.transform = 'translateY(0)';
            learnMoreCTA.style.pointerEvents = 'auto';
            learnMoreCTA.style.boxShadow = '0 0 20px rgba(46, 211, 198, 0.15)';
            learnMoreCTA.style.textShadow = '0 0 10px rgba(46, 211, 198, 0.3)';
          }, 5000);
          
          // FLASH/PULSE EFFECT: Create intrigue with periodic flashes
          let flashCount = 0;
          const flashInterval = setInterval(() => {
            if (flashCount >= 3) {
              clearInterval(flashInterval);
              return;
            }
            
            // Quick flash to bright
            learnMoreCTA.style.transition = 'all 0.2s ease-out';
            learnMoreCTA.style.color = 'rgba(46, 211, 198, 1)';
            learnMoreCTA.style.borderColor = 'rgba(46, 211, 198, 0.6)';
            learnMoreCTA.style.boxShadow = '0 0 40px rgba(46, 211, 198, 0.4), 0 0 80px rgba(46, 211, 198, 0.2)';
            learnMoreCTA.style.textShadow = '0 0 25px rgba(46, 211, 198, 0.6)';
            
            // Return to subtle after flash
            setTimeout(() => {
              learnMoreCTA.style.transition = 'all 0.6s ease-in';
              learnMoreCTA.style.color = 'rgba(46, 211, 198, 0.7)';
              learnMoreCTA.style.borderColor = 'rgba(46, 211, 198, 0.2)';
              learnMoreCTA.style.boxShadow = '0 0 20px rgba(46, 211, 198, 0.15)';
              learnMoreCTA.style.textShadow = '0 0 10px rgba(46, 211, 198, 0.3)';
            }, 200);
            
            flashCount++;
          }, 1500);
          
          // CONTINUOUS SUBTLE PULSE
          setTimeout(() => {
            const pulseKeyframes = `
              @keyframes deeperPulse {
                0%, 100% {
                  box-shadow: 0 0 20px rgba(46, 211, 198, 0.15);
                  text-shadow: 0 0 10px rgba(46, 211, 198, 0.3);
                }
                50% {
                  box-shadow: 0 0 30px rgba(46, 211, 198, 0.25);
                  text-shadow: 0 0 15px rgba(46, 211, 198, 0.4);
                }
              }
            `;
            
            const styleSheet = document.createElement('style');
            styleSheet.textContent = pulseKeyframes;
            document.head.appendChild(styleSheet);
            
            learnMoreCTA.style.animation = 'deeperPulse 3s ease-in-out infinite';
          }, 10000);
          
          // Hide CTA when animation completes (goes back up)
          setTimeout(() => {
            learnMoreCTA.style.opacity = '0';
            learnMoreCTA.style.transform = 'translateY(-10px)';
            setTimeout(() => {
              if (learnMoreCTA.parentNode) {
                learnMoreCTA.remove();
              }
            }, 400);
          }, 12000);
