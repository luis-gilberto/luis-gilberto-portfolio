5) 🧠 SUBTLE MOTION ENGINE — JS
======================================================
🔎 WHAT TO LOOK FOR
- Your small utilities JS or right before </body> on the HUB page.

✅ PASTE:

<script>
(function(){
  const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const isTouch = matchMedia('(hover: none), (pointer: coarse)').matches;

  // Show the eyebrow chip only when ?source=imcservices
  try {
    const params = new URLSearchParams(window.location.search);
    const chip = document.querySelector('[data-chip]');
    if (chip) chip.style.display = (params.get('source') === 'imcservices') ? 'inline-block' : 'none';
  } catch(e){}

  // Scroll reveal
  try {
    const obs = new IntersectionObserver((entries)=>{
      for (const e of entries){
        if (e.isIntersecting) {
          e.target.classList.add('is-in');
          obs.unobserve(e.target);
        }
      }
    }, { threshold: 0.2 });
    document.querySelectorAll('.lg-tilt').forEach(el => obs.observe(el));
  } catch(e){}

  // Micro tilt + gloss (desktop only + not reduced)
  if (!prefersReduced && !isTouch){
    const maxTilt = 6; // degrees
    const damp = 0.12; // easing

    document.querySelectorAll('.lg-tilt').forEach(card=>{
      let rx = 0, ry = 0;

      function onMove(ev){
        const r = card.getBoundingClientRect();
        const mx = ev.clientX - r.left;
        const my = ev.clientY - r.top;

        // Shine position
        card.style.setProperty('--mx', (mx / r.width) * 100 + '%');
        card.style.setProperty('--my', (my / r.height) * 100 + '%');

        // Target rotation
        const ty = ((mx / r.width) - 0.5) * (maxTilt * 2);  // rotateY
        const tx = -((my / r.height) - 0.5) * (maxTilt * 2); // rotateX

        // Ease toward target
        rx += (tx - parseFloat(rx)) * damp;
        ry += (ty - parseFloat(ry)) * damp;

        card.style.setProperty('--rx', rx.toFixed(2) + 'deg');
        card.style.setProperty('--ry', ry.toFixed(2) + 'deg');
      }

      function onLeave(){
        card.style.setProperty('--rx', '0deg');
        card.style.setProperty('--ry', '0deg');
        card.style.setProperty('--mx', '50%');
        card.style.setProperty('--my', '30%');
      }

      card.addEventListener('mousemove', onMove);
      card.addEventListener('mouseleave', onLeave);
    });
  }
})();
</script>
