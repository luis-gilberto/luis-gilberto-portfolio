    <!-- AUTOMATION SCRIPT -->
    <script>
    (function(){
        var target = document.getElementById('topics-page-content');
        
        // Fetch the existing metadata file
        fetch('/insights/data/articles-metadata.json')
            .then(r => r.json())
            .then(function(items){
                var map = new Map();
                
                // Group articles by topic
                items.forEach(function(a){
                    (a.topics || []).forEach(function(t){
                        if(!map.has(t)) map.set(t, []);
                        map.get(t).push(a);
                    })
                });

                // Sort topics alphabetically
                var keys = Array.from(map.keys()).sort();
                var html = '';

                // Generate HTML for each Topic Card
                keys.forEach(function(t){
                    // Sort articles by date within topic
                    var links = (map.get(t)||[]).sort(function(a,b){
                        return new Date(b.publishDate)-new Date(a.publishDate)
                    }).map(function(a){
                        // The New Link Structure
                        return '<a href="'+a.url+'">'+a.title+'</a>'
                    }).join('');

                    // The New Card Structure
                    html += `
                        <div class="topic-card">
                            <div class="topic-title">${t}</div>
                            <div class="topic-links">${links}</div>
                        </div>
                    `;
                });

                // Inject into page
                target.innerHTML = html;
            })
            .catch(function(err){
                target.innerHTML = '<div class="loading-text">Unable to load topics. Please try again later.</div>';
                console.error(err);
            });
    })();
    </script>
