  <script>
    (function() {
      function trackSignal(eventName, properties) {
        if (typeof plausible !== 'undefined') { plausible(eventName, { props: properties }); }
      }

      const STORAGE_KEY = 'lg_parent_lang';
      let currentLang = localStorage.getItem(STORAGE_KEY) || 'es';

      function applyLang(lang) {
        currentLang = lang;
        localStorage.setItem(STORAGE_KEY, lang);
        document.documentElement.lang = lang;
        
        document.querySelectorAll('[data-i18n]').forEach(el => {
          const txt = el.getAttribute(`data-${lang}`);
          if (txt) el.innerHTML = txt; 
        });

        document.querySelectorAll('.lang-btn').forEach(btn => {
          btn.setAttribute('aria-pressed', btn.getAttribute('data-lang') === lang ? 'true' : 'false');
        });
      }

      document.addEventListener('click', (e) => {
        const btn = e.target.closest('.lang-btn');
        if (btn) {
          applyLang(btn.getAttribute('data-lang'));
          trackSignal('language_changed', { to_lang: currentLang, location: 'portal' });
        }
      });

      applyLang(currentLang);

      const enterBtn = document.getElementById('enter-btn');
      if(enterBtn) {
        enterBtn.addEventListener('click', function(e) {
          e.preventDefault();
          const href = this.getAttribute('href');
          
          trackSignal('portal_entered', { language: currentLang });
          
          document.body.classList.add('page-exiting');
          
          setTimeout(() => {
            window.location.href = href;
          }, 800); 
        });
      }
    })();
  </script>
