D) GLOBAL JS HELPERS (add once near bottom before </body> of each page)
================================================================================
<script>
window.LG = window.LG || {};
LG.bridge = {
  addUTM(url, utm){ try{
    const u = new URL(url, window.location.origin);
    Object.entries(utm || {}).forEach(([k,v]) => v && u.searchParams.set(k,v));
    return u.toString();
  }catch(_){ return url; } },
  clickTrack(e){
    const card = e.target.closest('[data-bridge]'); if(!card) return;
    const { caseId, placement } = card.dataset;
    // lightweight event placeholder; swap to GA or CF analytics if needed
    console.debug('bridge:click', {caseId, placement});
  }
};
document.addEventListener('click', LG.bridge.clickTrack, {passive:true});
</script>
