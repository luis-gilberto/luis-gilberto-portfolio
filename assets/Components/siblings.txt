  <script>
    (function(){
      // SnglIQ TELEMETRY SAFE WRAPPER
      function trackSignal(eventName, properties) {
        if (typeof plausible !== 'undefined') {
          plausible(eventName, { props: properties });
        }
        console.log(`[SnglIQ Captured]: ${eventName}`, properties);
      }

      // STATE & UTILS
      let lang = localStorage.getItem('lg_parent_lang') || 'es';
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toast-msg');
      let toastTimer;

      function showToast(msg) {
        clearTimeout(toastTimer);
        toastMsg.textContent = msg;
        toast.classList.add('show');
        toastTimer = setTimeout(() => toast.classList.remove('show'), 3500);
      }

      // LANGUAGE TRANSLATOR SYSTEM
      function applyLang(l) {
        lang = l;
        document.documentElement.lang = l;
        localStorage.setItem('lg_parent_lang', l);
        
        document.querySelectorAll('[data-i18n]').forEach(el => {
          const t = el.getAttribute('data-' + l);
          if (t) el.innerHTML = t;
        });

        document.querySelectorAll('.lang-btn').forEach(btn => {
          btn.setAttribute('aria-pressed', btn.dataset.lang === l ? 'true' : 'false');
        });
      }

      // EVENT LISTENER: LANGUAGE
      document.querySelectorAll('.lang-btn').forEach(btn => {
        btn.addEventListener('click', (e) => {
          applyLang(e.target.dataset.lang);
        });
      });

      // EVENT LISTENER: COPY PHRASES (Tactile UX + Telemetry)
      document.querySelectorAll('.phrase-card').forEach(card => {
        card.addEventListener('click', () => {
          const phrase = card.querySelector('.pc-phrase').textContent.trim();
          const mode = card.getAttribute('data-mode');

          navigator.clipboard.writeText(phrase).then(() => {
            // Revert others
            document.querySelectorAll('.phrase-card').forEach(c => {
              c.setAttribute('aria-pressed', 'false');
              const cp = c.querySelector('.pc-copy');
              cp.textContent = cp.getAttribute('data-' + lang) || (lang === 'es' ? 'Copiar' : 'Copy');
            });

            // Activate current
            card.setAttribute('aria-pressed', 'true');
            card.querySelector('.pc-copy').textContent = lang === 'es' ? '¡Copiado!' : 'Copied!';
            
            showToast(lang === 'es' ? '¡Copiado! Úsalo ahora.' : 'Copied! Use it now.');

            // SnglIQ Trigger (FIXED to sibling_conflict)
            trackSignal('phrase_copied', { mode: mode, language: lang, protocol: 'sibling_conflict' });

            setTimeout(() => {
              card.setAttribute('aria-pressed', 'false');
              const cp = card.querySelector('.pc-copy');
              cp.textContent = cp.getAttribute('data-' + lang) || (lang === 'es' ? 'Copiar' : 'Copy');
            }, 3000);
          });
        });
      });

      // EVENT LISTENER: ACCORDIONS (Telemetry)
      document.querySelectorAll('.accordion').forEach(acc => {
        const trigger = acc.querySelector('.acc-trig');
        const topic = acc.getAttribute('data-topic');

        trigger.addEventListener('click', () => {
          const isOpen = acc.hasAttribute('open');
          acc.toggleAttribute('open', !isOpen);
          
          if (!isOpen) { // Means it was just opened
            trackSignal('card_opened', { topic: topic, protocol: 'sibling_conflict' });
          }
        });
      });

      // EVENT LISTENER: CROSS PROTOCOL (Telemetry)
      const crossLink = document.querySelector('.cp-link');
      if(crossLink) {
        crossLink.addEventListener('click', () => {
          trackSignal('cross_nav', { from: 'sibling_conflict', to: 'emotional_escalation' });
        });
      }

      // EVENT LISTENER: SHARING (Telemetry)
      function getShareData() {
        return {
          title: document.title,
          text: lang === 'es' ? 'Para tener a la mano. Protocolo de Criar Sin Culpas para conflictos entre hermanos: ' : 'Keep this handy. Sibling conflict protocol: ',
          url: window.location.href.split('?')[0]
        };
      }

      const btnWa = document.querySelector('.share-wa');
      if(btnWa) {
        btnWa.addEventListener('click', () => {
          const d = getShareData();
          trackSignal('share_whatsapp', { protocol: 'sibling_conflict', context: 'caregiver_alignment' });
          window.open('https://wa.me/?text=' + encodeURIComponent(d.text + d.url), '_blank');
        });
      }

      const btnCopy = document.querySelector('.share-copy');
      if(btnCopy) {
        btnCopy.addEventListener('click', () => {
          const d = getShareData();
          navigator.clipboard.writeText(d.url);
          trackSignal('share_copy_link', { protocol: 'sibling_conflict' });
          showToast(lang === 'es' ? '¡Enlace copiado!' : 'Link copied!');
        });
      }

      // Initialize Language
      applyLang(lang);

    })();
  </script>
