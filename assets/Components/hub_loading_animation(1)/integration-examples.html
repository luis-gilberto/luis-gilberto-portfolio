The Hub Loading Animation

Production-ready integration examples and setup guide

🚀 Basic Integration

Simplest way to add the loading animation to your website:

<!-- Add to your HTML head -->
<script src="assets/js/hub-loader.min.js"></script>

<!-- Initialize on page load -->
<script>
document.addEventListener('DOMContentLoaded', () => {
    HubLoader.create({
        onComplete: () => {
            // Show your main content
            document.getElementById('main-content').style.display = 'block';
        }
    });
});
</script>

⚙️ Advanced Configuration

Full customization with all options:

const loader = HubLoader.create({
    // Video settings
    videoUrl: './assets/coral_laser_line.mp4',
    
    // Timing
    duration: 8000, // 8 seconds total
    
    // UI Options
    showProgress: true,
    skipOnClick: true,
    
    // Callbacks
    onComplete: () => {
        showMainContent();
        trackEvent('loading_complete');
    }
});

⚛️ React Integration

Perfect for React/Next.js applications:

import { useEffect, useState } from 'react';

function App() {
    const [loading, setLoading] = useState(true);
    
    useEffect(() => {
        if (typeof window !== 'undefined') {
            const { HubLoader } = require('./hub-loader.min.js');
            
            HubLoader.create({
                onComplete: () => setLoading(false)
            });
        }
    }, []);
    
    if (loading) return null;
    
    return <YourMainApp />;
}

🔌 WordPress Integration

Easy integration with WordPress themes:

<?php
// In your theme's functions.php
function enqueue_hub_loader() {
    wp_enqueue_script(
        'hub-loader',
        get_template_directory_uri() . '/assets/js/hub-loader.min.js',
        array(),
        '1.0.0',
        true
    );
}
add_action('wp_enqueue_scripts', 'enqueue_hub_loader');
?>

<!-- In your template -->
<script>
jQuery(document).ready(() => {
    HubLoader.create({
        onComplete: () => {
            jQuery('#main-content').fadeIn();
        }
    });
});
</script>

⚡ Performance Optimization

Best practices for lightning-fast loading:

// Preload video for instant playback
<link rel="preload" href="assets/coral_laser_line.mp4" as="video" type="video/mp4">

// Lazy load main content while animation plays
const loader = HubLoader.create({
    onComplete: () => {
        // Content is loaded by now
        showContent();
    }
});

// Load your heavy assets in parallel
Promise.all([
    loadImages(),
    loadScripts(),
    loadData()
]).then(() => {
    console.log('All assets loaded');
});

📊 Analytics Integration

Track loading performance and user engagement:

const startTime = performance.now();

HubLoader.create({
    onComplete: () => {
        const loadTime = performance.now() - startTime;
        
        // Google Analytics 4
// Custom analytics
        analytics.track('Hub Loading Complete', {
            duration: loadTime,
            device: isMobile() ? 'mobile' : 'desktop'
        });
    }
});
