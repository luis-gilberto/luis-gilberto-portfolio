<script>
  // Immersive scroll animation
  (function initImmersive(){
    const hero = document.getElementById('immersiveHero');
    if(!hero) return;
    const imgs = hero.querySelectorAll('.hero__img');
    const endcap = hero.querySelector('.hero__endcap');
    
    function onScroll(){
      const rect = hero.getBoundingClientRect();
      const vh = window.innerHeight;
      const scrollHeight = hero.scrollHeight;
      const progress = Math.max(0, Math.min(1, (vh - rect.top) / vh));
      
      imgs.forEach(img=>{
        const maxScale = parseFloat(getComputedStyle(img).getPropertyValue('--maxScale')) || 6;
        const scale = 1 + progress * (maxScale - 1);
        const alpha = 1 - progress * 1.2;
        img.style.setProperty('--scale', scale.toFixed(3));
        img.style.setProperty('--alpha', Math.max(0, alpha).toFixed(3));
      });
      
      const endStart = 0.7;
      const endProgress = Math.max(0, Math.min(1, (progress - endStart)/(1 - endStart)));
      endcap.style.setProperty('--endScale', (0.8 + 0.2*endProgress).toFixed(3));
      endcap.style.setProperty('--endAlpha', endProgress.toFixed(3));
    }
    
    window.addEventListener('scroll', onScroll, {passive:true});
    onScroll();
  })();

  // Mood selector functionality
  const TRACK_ORDERS = {
    recruiter:['timeline','projects','how','essays','contact'],
    partner:['projects','how','timeline','essays','contact'],
    inspiration:['projects','essays','timeline','how','contact'],
    default:['projects','timeline','how','essays','contact']
  };
  const CTA_TEXT = {
    recruiter:{explore:'Case Studies (Top 6)', timeline:'Timeline & Roles'},
    partner:{explore:'Campaigns & Toolkits', timeline:'How I Work'},
    inspiration:{explore:'Visual Gallery', timeline:'Story Timeline'},
    default:{explore:'Explore Projects', timeline:'View Timeline'}
  };
  
  function applyTrack(track){
    const chosen = TRACK_ORDERS[track] ? track : 'default';
    localStorage.setItem('lg_track', chosen);
    
    const content = document.getElementById('pageContent');
    if(content){
      const map = {};
      [...content.children].forEach(sec => map[sec.dataset.section] = sec);
      TRACK_ORDERS[chosen].forEach(key => { if(map[key]) content.appendChild(map[key]); });
    }
    
    const explore = document.querySelector('[data-cta="explore"]');
    const timeline = document.querySelector('[data-cta="timeline"]');
    if(explore) explore.textContent = CTA_TEXT[chosen].explore;
    if(timeline) timeline.textContent = CTA_TEXT[chosen].timeline;
  }
  
  // Event listeners
  document.getElementById('trackSelector').addEventListener('click', (e)=>{
    const card = e.target.closest('.selector__card');
    if(!card) return;
    applyTrack(card.dataset.track);
    document.getElementById('pageContent')?.scrollIntoView({behavior:'smooth'});
  });
  
  // Initialize
  const saved = localStorage.getItem('lg_track') || 'default';
  applyTrack(saved);
</script>
