    <script>
        // Pricing password reveal
        (function(){
            const btn = document.getElementById('revealPriceBtn');
            const pwd = document.getElementById('pricePassword');
            const status = document.getElementById('priceStatus');
            function reveal() {
                const val = (pwd && pwd.value ? pwd.value.trim() : '');
                if (val === 'LG100') {
                    document.querySelectorAll('.tier-price').forEach(el => el.classList.add('revealed'));
                    if (status) { 
                        status.textContent = 'Pricing revealed'; 
                        status.classList.remove('error'); 
                        status.classList.add('ok'); 
                    }
                } else {
                    if (status) { 
                        status.textContent = 'Incorrect password'; 
                        status.classList.remove('ok'); 
                        status.classList.add('error'); 
                    }
                }
            }
            if (btn) btn.addEventListener('click', reveal);
            if (pwd) pwd.addEventListener('keyup', function(e){ 
                if (e.key === 'Enter') reveal(); 
            });
        })();
    </script>

    <script>
        // Portfolio Bridge helpers (scoped): analytics + keyboard accessibility
        (function(){
            function track(action, label){ 
                try { 
                    if (window.gtag) window.gtag('event', action, { event_label: label }); 
                } catch(e){} 
            }
            document.querySelectorAll('.portfolio-bridge .bridge-card a').forEach(function(a){
                var card = a.closest('.bridge-card'); 
                if (!card) return;
                card.setAttribute('tabindex','0');
                card.addEventListener('keydown', function(e){ 
                    if (e.key==='Enter'||e.key===' ') { 
                        e.preventDefault(); 
                        a.click(); 
                    }
                });
                a.addEventListener('click', function(){ 
                    track('bridge_click', a.getAttribute('href')); 
                });
            });
        })();
    </script>
