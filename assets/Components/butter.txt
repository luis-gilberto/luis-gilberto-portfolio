1. SMOOTH SCROLLING JAVASCRIPT IMPLEMENTATION
================================================================

Add this complete JavaScript code to enable smooth section-to-section scrolling:

/* Enhanced Smooth Section Scrolling - Complete Implementation */
class SmoothSectionScroller {
    constructor() {
        this.sections = [];
        this.currentSection = 0;
        this.isScrolling = false;
        this.scrollTimeout = null;
        this.touchStartY = 0;
        this.scrollDirection = null;
        
        this.init();
    }
    
    init() {
        this.setupSections();
        this.bindEvents();
        this.updateActiveSection();
    }
    
    setupSections() {
        // Get all main sections
        this.sections = Array.from(document.querySelectorAll('section[id], .hero[id]'));
        
        // Ensure each section takes full viewport height
        this.sections.forEach((section, index) => {
            section.style.minHeight = '100vh';
            section.style.display = 'flex';
            section.style.flexDirection = 'column';
            section.style.justifyContent = 'center';
            section.setAttribute('data-section-index', index);
        });
    }
    
    bindEvents() {
        // Mouse wheel events
        document.addEventListener('wheel', this.handleWheel.bind(this), { passive: false });
        
        // Touch events for mobile
        document.addEventListener('touchstart', this.handleTouchStart.bind(this), { passive: true });
        document.addEventListener('touchend', this.handleTouchEnd.bind(this), { passive: false });
        
        // Keyboard events
        document.addEventListener('keydown', this.handleKeydown.bind(this));
        
        // Window resize
        window.addEventListener('resize', this.debounce(this.handleResize.bind(this), 250));
        
        // Update on hash change
        window.addEventListener('hashchange', this.handleHashChange.bind(this));
    }
    
    handleWheel(e) {
        if (this.isScrolling) {
            e.preventDefault();
            return;
        }
        
        e.preventDefault();
        
        const delta = e.deltaY;
        const direction = delta > 0 ? 'down' : 'up';
        
        this.scrollToSection(direction);
    }
    
    handleTouchStart(e) {
        this.touchStartY = e.touches[0].clientY;
    }
    
    handleTouchEnd(e) {
        if (this.isScrolling) {
            e.preventDefault();
            return;
        }
        
        const touchEndY = e.changedTouches[0].clientY;
        const deltaY = this.touchStartY - touchEndY;
        
        // Minimum swipe distance
        if (Math.abs(deltaY) < 50) return;
        
        e.preventDefault();
        
        const direction = deltaY > 0 ? 'down' : 'up';
        this.scrollToSection(direction);
    }
    
    handleKeydown(e) {
        if (this.isScrolling) return;
        
        switch(e.key) {
            case 'ArrowDown':
            case 'PageDown':
            case ' ':
                e.preventDefault();
                this.scrollToSection('down');
                break;
            case 'ArrowUp':
            case 'PageUp':
                e.preventDefault();
                this.scrollToSection('up');
                break;
            case 'Home':
                e.preventDefault();
                this.scrollToSectionIndex(0);
                break;
            case 'End':
                e.preventDefault();
                this.scrollToSectionIndex(this.sections.length - 1);
                break;
        }
    }
    
    scrollToSection(direction) {
        let targetIndex = this.currentSection;
        
        if (direction === 'down' && this.currentSection < this.sections.length - 1) {
            targetIndex = this.currentSection + 1;
        } else if (direction === 'up' && this.currentSection > 0) {
            targetIndex = this.currentSection - 1;
        }
        
        if (targetIndex !== this.currentSection) {
            this.scrollToSectionIndex(targetIndex);
        }
    }
    
    scrollToSectionIndex(index) {
        if (index < 0 || index >= this.sections.length || this.isScrolling) return;
        
        this.isScrolling = true;
        this.currentSection = index;
        
        const targetSection = this.sections[index];
        const targetPosition = targetSection.offsetTop;
        
        // Update URL hash
        if (targetSection.id) {
            history.replaceState(null, null, `#${targetSection.id}`);
        }
        
        // Smooth scroll to target
        this.smoothScrollTo(targetPosition, () => {
            this.isScrolling = false;
            this.updateActiveSection();
        });
        
        // Update TOC if it exists
        this.updateTOC();
    }
    
    smoothScrollTo(targetY, callback) {
        const startY = window.pageYOffset;
        const distance = targetY - startY;
        const duration = Math.min(Math.abs(distance) * 0.5, 1000); // Max 1 second
        const startTime = Date.now();
        
        const easeInOutCubic = (t) => {
            return t < 0.5 ? 4 * t * t * t : (t - 1) * (2 * t - 2) * (2 * t - 2) + 1;
        };
        
        const scroll = () => {
            const elapsed = Date.now() - startTime;
            const progress = Math.min(elapsed / duration, 1);
            const easedProgress = easeInOutCubic(progress);
            
            window.scrollTo(0, startY + distance * easedProgress);
            
            if (progress < 1) {
                requestAnimationFrame(scroll);
            } else if (callback) {
                callback();
            }
        };
        
        requestAnimationFrame(scroll);
    }
    
    updateActiveSection() {
        // Update section classes
        this.sections.forEach((section, index) => {
            if (index === this.currentSection) {
                section.classList.add('active-section');
            } else {
                section.classList.remove('active-section');
            }
        });
    }
    
    updateTOC() {
        // Update Table of Contents if it exists
        const tocLinks = document.querySelectorAll('.toc-link');
        if (tocLinks.length === 0) return;
        
        const activeSection = this.sections[this.currentSection];
        if (!activeSection || !activeSection.id) return;
        
        tocLinks.forEach(link => {
            const href = link.getAttribute('href');
            if (href === `#${activeSection.id}`) {
                link.classList.add('active');
            } else {
                link.classList.remove('active');
            }
        });
    }
    
    handleHashChange() {
        const hash = window.location.hash.substring(1);
        if (!hash) return;
        
        const targetSection = document.getElementById(hash);
        if (!targetSection) return;
        
        const index = this.sections.indexOf(targetSection);
        if (index !== -1 && index !== this.currentSection) {
            this.scrollToSectionIndex(index);
        }
    }
    
    handleResize() {
        // Recalculate section positions on resize
        this.setupSections();
        
        // Scroll to current section to maintain position
        if (this.sections[this.currentSection]) {
            const targetPosition = this.sections[this.currentSection].offsetTop;
            window.scrollTo(0, targetPosition);
        }
    }
    
    debounce(func, wait) {
        let timeout;
        return function executedFunction(...args) {
            const later = () => {
                clearTimeout(timeout);
                func(...args);
            };
            clearTimeout(timeout);
            timeout = setTimeout(later, wait);
        };
    }
    
    // Public methods for external control
    goToSection(sectionId) {
        const section = document.getElementById(sectionId);
        if (!section) return;
        
        const index = this.sections.indexOf(section);
        if (index !== -1) {
            this.scrollToSectionIndex(index);
        }
    }
    
    next() {
        this.scrollToSection('down');
    }
    
    previous() {
        this.scrollToSection('up');
    }
}

// Initialize smooth scrolling when DOM is ready
document.addEventListener('DOMContentLoaded', function() {
    // Only initialize if not on mobile (optional)
    const isMobile = window.innerWidth <= 768;
    const enableOnMobile = true; // Set to false if you want to disable on mobile
    
    if (!isMobile || enableOnMobile) {
        window.smoothScroller = new SmoothSectionScroller();
    }
});

Additional JavaScript for enhanced touch/trackpad support:

/* Enhanced Touch Support */
class EnhancedTouchHandler {
    constructor(smoothScroller) {
        this.smoothScroller = smoothScroller;
        this.touchStartY = 0;
        this.touchStartTime = 0;
        this.velocityTracker = [];
        
        this.bindTouchEvents();
    }
    
    bindTouchEvents() {
        document.addEventListener('touchstart', this.handleTouchStart.bind(this), { passive: true });
        document.addEventListener('touchmove', this.handleTouchMove.bind(this), { passive: true });
        document.addEventListener('touchend', this.handleTouchEnd.bind(this), { passive: false });
    }
    
    handleTouchStart(e) {
        this.touchStartY = e.touches[0].clientY;
        this.touchStartTime = Date.now();
        this.velocityTracker = [];
    }
    
    handleTouchMove(e) {
        const currentY = e.touches[0].clientY;
        const currentTime = Date.now();
        
        this.velocityTracker.push({
            y: currentY,
            time: currentTime
        });
        
        // Keep only recent touch points
        if (this.velocityTracker.length > 5) {
            this.velocityTracker.shift();
        }
    }
    
    handleTouchEnd(e) {
        if (this.smoothScroller.isScrolling) {
            e.preventDefault();
            return;
        }
        
        const touchEndY = e.changedTouches[0].clientY;
        const touchEndTime = Date.now();
        const deltaY = this.touchStartY - touchEndY;
        const deltaTime = touchEndTime - this.touchStartTime;
        
        // Calculate velocity
        let velocity = 0;
        if (this.velocityTracker.length >= 2) {
            const recent = this.velocityTracker[this.velocityTracker.length - 1];
            const previous = this.velocityTracker[0];
            velocity = Math.abs((recent.y - previous.y) / (recent.time - previous.time));
        }
        
        // Determine if it's a swipe gesture
        const isSwipe = Math.abs(deltaY) > 30 && (velocity > 0.5 || Math.abs(deltaY) > 100);
        
        if (isSwipe) {
            e.preventDefault();
            const direction = deltaY > 0 ? 'down' : 'up';
            this.smoothScroller.scrollToSection(direction);
        }
    }
}

// Initialize enhanced touch support
document.addEventListener('DOMContentLoaded', function() {
    setTimeout(() => {
        if (window.smoothScroller) {
            new EnhancedTouchHandler(window.smoothScroller);
        }
    }, 100);
});

class PerformanceOptimizer {
    constructor() {
        this.isScrolling = false;
        this.scrollingClass = 'scrolling-active';
    }
    
    startScrolling() {
        if (!this.isScrolling) {
            document.body.classList.add(this.scrollingClass);
            this.isScrolling = true;
        }
    }
    
    endScrolling() {
        setTimeout(() => {
            document.body.classList.remove(this.scrollingClass);
            this.isScrolling = false;
        }, 300);
    }
}
